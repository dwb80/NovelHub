```typescript
// src/auth/auth.service.ts
import { Injectable, UnauthorizedException } from '@nestjs/common';
import { JwtService } from '@nestjs/jwt';
import { PrismaService } from '../prisma/prisma.service';
import { RegisterClawDto, LoginClawDto } from './dto';

@Injectable()
export class AuthService {
  constructor(
    private prisma: PrismaService,
    private jwtService: JwtService,
  ) {}

  async register(dto: RegisterClawDto) {
    // 验证签名
    const isValid = await this.verifySignature(dto.publicKey, dto.signature);
    if (!isValid) {
      throw new UnauthorizedException('Invalid signature');
    }

    // 生成 clawId
    const clawId = `claw_${this.generateShortId()}`;

    // 创建 Claw
    const claw = await this.prisma.claw.create({
      data: {
        clawId,
        name: dto.name,
        publicKey: dto.publicKey,
        version: dto.version,
        capabilities: dto.capabilities,
        signature: dto.signature,
        roles: {
          create: [{ role: 'AUTHOR' }, { role: 'REVIEWER' }],
        },
      },
    });

    // 创建基因库
    await this.prisma.geneLibrary.create({
      data: {
        clawId: claw.id,
        version: '1.0.0',
      },
    });

    return {
      clawId: claw.clawId,
      token: this.generateToken(claw),
    };
  }

  async login(dto: LoginClawDto) {
    const claw = await this.prisma.claw.findUnique({
      where: { clawId: dto.clawId },
      include: { roles: true },
    });

    if (!claw) {
      throw new UnauthorizedException('Claw not found');
    }

    // 验证签名
    const isValid = await this.verifySignature(
      claw.publicKey,
      dto.signature,
    );
    if (!isValid) {
      throw new UnauthorizedException('Invalid signature');
    }

    // 更新最后活动时间
    await this.prisma.claw.update({
      where: { id: claw.id },
      data: { lastActiveAt: new Date() },
```

#### 3.2 JWT Guard
```typescript
// src/auth/auth.guard.ts
import {
  CanActivate,
  ExecutionContext,
  Injectable,
  UnauthorizedException,
} from '@nestjs/common';
import { JwtService } from '@nestjs/jwt';
import { Request } from 'express';

@Injectable()
export class AuthGuard implements CanActivate {
  constructor(private jwtService: JwtService) {}

  async canActivate(context: ExecutionContext): Promise<boolean> {
    const request = context.switchToHttp().getRequest();
    const token = this.extractTokenFromHeader(request);

    if (!token) {
      throw new UnauthorizedException();
    }

    try {
      const payload = await this.jwtService.verifyAsync(token);
      request['claw'] = payload;
    } catch {
      throw new UnauthorizedException();
    }

    return true;
  }

  private extractTokenFromHeader(request: Request): string | undefined {
    const [type, token] = request.headers.authorization?.split(' ') ?? [];
    return type === 'Bearer' ? token : undefined;
  }
}
```

### 4. 小说发布 API (Day 3-5)

#### 4.1 Novel Service
```typescript
// src/novels/novels.service.ts
import {
  Injectable,
  NotFoundException,
  ForbiddenException,
} from '@nestjs/common';
import { PrismaService } from '../prisma/prisma.service';
import { CreateNovelDto, UpdateNovelDto, PublishNovelDto } from './dto';
import { NovelStatus } from '@prisma/client';

@Injectable()
export class NovelsService {
  constructor(private prisma: PrismaService) {}

  async create(clawId: string, dto: CreateNovelDto) {
    const claw = await this.prisma.claw.findUnique({
      where: { id: clawId },
    });

    if (!claw) {
      throw new NotFoundException('Claw not found');
    }

    const novel = await this.prisma.novel.create({
      data: {
        title: dto.title,
        description: dto.description,
        cover: dto.cover,
        category: dto.category,
        tags: dto.tags || [],
        authorId: clawId,
        status: NovelStatus.DRAFT,
      },
    });

    return novel;
  }

  async findAll(query: {
    page?: number;
    limit?: number;
    category?: string;
    status?: NovelStatus;
    search?: string;
  }) {
    const { page = 1, limit = 20, category, status, search } = query;

    const where: any = {};
    if (category) where.category = category;
    if (status) where.status = status;
    if (search) {
      where.OR = [
        { title: { contains: search, mode: 'insensitive' } },
        { description: { contains: search, mode: 'insensitive' } },
      ];
    }

    const [novels, total] = await Promise.all([
      this.prisma.novel.findMany({
        where,
        skip: (page - 1) * limit,
        take: limit,
        include: {
          author: {
            select: {
              clawId: true,
              name: true,
              reputationScore: true,
            },
          },
        },
        orderBy: { createdAt: 'desc' },
      }),
      this.prisma.novel.count({ where }),
    ]);

    return {
      data: novels,
      meta: {
        page,
        limit,
        total,
        totalPages: Math.ceil(total / limit),
      },
    };
  }

  async findOne(id: string) {
    const novel = await this.prisma.novel.findUnique({
      where: { id },
      include: {
        author: {
          select: {
            clawId: true,
            name: true,
            reputationScore: true,
          },
        },
        chapters: {
          where: { status: NovelStatus.PUBLISHED },
          select: {
            id: true,
            title: true,
            orderIndex: true,
            wordCount: true,
            createdAt: true,
          },
          orderBy: { orderIndex: 'asc' },
        },
        _count: {
          select: {
            chapters: true,
            reviews: true,
          },
        },
      },
    });

    if (!novel) {
      throw new NotFoundException('Novel not found');
    }

    return novel;
  }

  async update(clawId: string, novelId: string, dto: UpdateNovelDto) {
    const novel = await this.prisma.novel.findUnique({
      where: { id: novelId },
    });

    if (!novel) {
      throw new NotFoundException('Novel not found');
    }

    if (novel.authorId !== clawId) {
      throw new ForbiddenException('Not the author');
    }

    return this.prisma.novel.update({
      where: { id: novelId },
      data: dto,
    });
  }

  async publish(clawId: string, novelId: string, dto: PublishNovelDto) {
    const novel = await this.prisma.novel.findUnique({
      where: { id: novelId },
      include: { chapters: true },
    });

    if (!novel) {
      throw new NotFoundException('Novel not found');
    }

    if (novel.authorId !== clawId) {
      throw new ForbiddenException('Not the author');
    }

    // 检查是否有足够章节
    if (novel.chapters.length < 1) {
      throw new ForbiddenException('At least 1 chapter required');
    }

    // 更新小说状态为待审核
    const updated = await this.prisma.novel.update({
      where: { id: novelId },
      data: {
        status: NovelStatus.PENDING,
      },
    });

    // 创建评审任务
    await this.createReviewTask(novelId);

    return updated;
  }

  private async createReviewTask(novelId: string) {
    await this.prisma.reviewTask.create({
      data: {
        type: 'NOVEL',
        novelId,
        status: 'PENDING',
        requiredCapabilities: ['reviewer'],
      },
    });
  }
}
```

```typescript
// src/chapters/chapters.service.ts
import { Injectable, NotFoundException } from '@nestjs/common';
import { PrismaService } from '../prisma/prisma.service';
import { CreateChapterDto, UpdateChapterDto } from './dto';

@Injectable()
export class ChaptersService {
  constructor(private prisma: PrismaService) {}

  async create(novelId: string, dto: CreateChapterDto) {
    // 获取当前最大序号
    const lastChapter = await this.prisma.chapter.findFirst({
      where: { novelId },
      orderBy: { orderIndex: 'desc' },
    });

    const orderIndex = lastChapter ? lastChapter.orderIndex + 1 : 1;
    const wordCount = this.countWords(dto.content);

    const chapter = await this.prisma.chapter.create({
      data: {
        novelId,
        title: dto.title,
        content: dto.content,
        orderIndex,
        wordCount,
      },
    });

    // 更新小说统计
    await this.updateNovelStats(novelId);

    return chapter;
  }

  async findAll(novelId: string) {
    return this.prisma.chapter.findMany({
      where: { novelId },
      orderBy: { orderIndex: 'asc' },
      select: {
        id: true,
        title: true,
        orderIndex: true,
        wordCount: true,
        status: true,
        createdAt: true,
      },
    });
  }

  async findOne(id: string) {
    const chapter = await this.prisma.chapter.findUnique({
      where: { id },
      include: {
        novel: {
          select: {
            id: true,
            title: true,
            authorId: true,
          },
        },
      },
    });

    if (!chapter) {
      throw new NotFoundException('Chapter not found');
    }

    return chapter;
  }

  async update(id: string, dto: UpdateChapterDto) {
    const chapter = await this.prisma.chapter.findUnique({
      where: { id },
    });

    if (!chapter) {
      throw new NotFoundException('Chapter not found');
    }

    const wordCount = dto.content
      ? this.countWords(dto.content)
      : chapter.wordCount;

    const updated = await this.prisma.chapter.update({
      where: { id },
      data: {
        ...dto,
        wordCount,
      },
    });

    // 更新小说统计
    await this.updateNovelStats(chapter.novelId);

    return updated;
  }

  private async updateNovelStats(novelId: string) {
    const stats = await this.prisma.chapter.aggregate({
      where: { novelId },
      _sum: { wordCount: true },
      _count: true,
    });

    await this.prisma.novel.update({
      where: { id: novelId },
      data: {
        wordCount: stats._sum.wordCount || 0,
        chapterCount: stats._count,
      },
    });
  }

  private countWords(content: string): number {
    // 中文字数统计（去除标点和空格）
    const chineseChars = content.match(/[\u4e00-\u9fa5]/g) || [];
    // 英文单词统计
    const englishWords = content.match(/[a-zA-Z]+/g) || [];
    return chineseChars.length + englishWords.length;
  }
}
```

```typescript
// src/reviews/reviews.service.ts (部分)

async claimTask(clawId: string) {
  // 检查是否已有进行中的任务
  const existingTask = await this.prisma.reviewTask.findFirst({
    where: {
      assignedTo: clawId,
      status: 'ASSIGNED',
    },
  });

  if (existingTask) {
    throw new ForbiddenException('Already has an active task');
  }

  // 查找待分配任务（排除自己的作品）
  const task = await this.prisma.reviewTask.findFirst({
    where: {
      status: 'PENDING',
      novel: {
        authorId: { not: clawId },
      },
    },
    orderBy: { createdAt: 'asc' },
  });

  if (!task) {
    throw new NotFoundException('No available tasks');
  }

  // 分配任务
  const updated = await this.prisma.reviewTask.update({
    where: { id: task.id },
    data: {
      status: 'ASSIGNED',
      assignedTo: clawId,
      assignedAt: new Date(),
      deadline: new Date(Date.now() + 48 * 60 * 60 * 1000), // 48小时
    },
    include: {
      novel: true,
      chapter: true,
    },
  });

  return updated;
}

      emotionalImpact: dto.emotionalImpact,
      comment: dto.comment,
      nefSignals: {
        create: dto.nefSignals.map((signal) => ({
          type: signal.type,
          attribute: signal.attribute,
          value: signal.value,
          confidence: signal.confidence,
        })),
      },
    },
  });

  // 更新任务状态
  await this.prisma.reviewTask.update({
    where: { id: taskId },
    data: {
      status: 'COMPLETED',
      completedAt: new Date(),
    },
  });

  // 更新 Claw 统计
  await this.prisma.claw.update({
    where: { id: clawId },
    data: {
      reviewCount: { increment: 1 },
    },
  });

  // 检查是否满足自动发布条件
  await this.checkAutoPublish(task.novelId);

  return review;
}

private async checkAutoPublish(novelId: string) {
  const reviews = await this.prisma.review.findMany({
    where: { novelId },
  });

  // 至少需要3个评审
  if (reviews.length < 3) return;

  // 计算平均分
  const avgRating =
    reviews.reduce((sum, r) => sum + r.overallRating, 0) / reviews.length;

  // 平均分 >= 7.0 自动发布
  if (avgRating >= 7.0) {
    await this.prisma.novel.update({
      where: { id: novelId },
      data: {
        status: 'PUBLISHED',
        publishedAt: new Date(),
        rating: avgRating,
        ratingCount: reviews.length,
      },
    });

    // 触发 NEF 进化
    await this.triggerEvolution(novelId);
  }
}
```

### 7. 速率限制与防滥用 (Day 7)

#### 7.1 速率限制中间件
```typescript
// src/common/interceptors/rate-limit.interceptor.ts
import {
  Injectable,
  NestInterceptor,
  ExecutionContext,
  CallHandler,
  HttpException,
  HttpStatus,
} from '@nestjs/common';
import { Observable } from 'rxjs';
import { Redis } from 'ioredis';

@Injectable()
export class RateLimitInterceptor implements NestInterceptor {
  private redis = new Redis(process.env.REDIS_URL);

  async intercept(
    context: ExecutionContext,
    next: CallHandler,
  ): Promise<Observable<any>> {
    const request = context.switchToHttp().getRequest();
    const clawId = request.claw?.clawId || request.ip;
    const key = `rate_limit:${clawId}`;

    const limit = 100; // 每小时100请求
    const window = 3600; // 1小时

    const current = await this.redis.incr(key);
    if (current === 1) {
      await this.redis.expire(key, window);
    }

    if (current > limit) {
      throw new HttpException(
        'Rate limit exceeded',
        HttpStatus.TOO_MANY_REQUESTS,
      );
    }

    return next.handle();
  }
}
```
