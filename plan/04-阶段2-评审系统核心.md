```typescript
// src/reviews/services/task-matcher.service.ts
import { Injectable } from '@nestjs/common';
import { PrismaService } from '../../prisma/prisma.service';
import { ReviewTask, Claw, TaskType } from '@prisma/client';

interface MatchScore {
  taskId: string;
  score: number;
  reasons: string[];
}

@Injectable()
export class TaskMatcherService {
  constructor(private prisma: PrismaService) {}

  /**
   * 为 Claw 匹配最佳评审任务
   */
  async findBestMatch(clawId: string): Promise<ReviewTask | null> {
    const claw = await this.prisma.claw.findUnique({
      where: { id: clawId },
      include: { 
        geneLibrary: {
          include: {
            plotGenes: true,
            characterCapsules: true,
            styleModules: true,
          }
        }
      }
    });

    if (!claw) return null;

    // 获取待分配任务
    const pendingTasks = await this.prisma.reviewTask.findMany({
      where: {
        status: 'PENDING',
        novel: {
          authorId: { not: clawId }, // 排除自己的作品
        },
      },
      include: {
        novel: {
          include: {
            chapters: {
              where: { status: 'PUBLISHED' },
              take: 3,
            },
          },
        },
      },
      take: 50, // 评估最近50个任务
    });

    if (pendingTasks.length === 0) return null;

    // 计算匹配分数
    const scores: MatchScore[] = pendingTasks.map(task => ({
      taskId: task.id,
      score: this.calculateMatchScore(claw, task),
      reasons: this.getMatchReasons(claw, task),
    }));

    // 排序并选择最佳匹配
    scores.sort((a, b) => b.score - a.score);
    const bestMatch = scores[0];

    if (bestMatch.score < 0.3) {
      // 分数太低，使用轮询策略
      return this.getRoundRobinTask(clawId, pendingTasks);
    }

    return pendingTasks.find(t => t.id === bestMatch.taskId) || null;
  }

  /**
   * 计算匹配分数 (0-1)
   */
  private calculateMatchScore(claw: any, task: ReviewTask & { novel: any }): number {
    let score = 0;
    let weights = 0;

    // 1. 能力标签匹配 (权重: 0.3)
    const capabilityMatch = this.calculateCapabilityMatch(
      claw.capabilities,
      task.requiredCapabilities
    );
    score += capabilityMatch * 0.3;
    weights += 0.3;

    // 2. 小说类型专长 (权重: 0.25)
    const genreMatch = this.calculateGenreMatch(claw, task.novel.category);
    score += genreMatch * 0.25;
    weights += 0.25;

    // 3. 历史评审质量 (权重: 0.2)
    const qualityScore = claw.reputationScore / 100;
    score += qualityScore * 0.2;
    weights += 0.2;

    // 4. 工作负载平衡 (权重: 0.15)
    const workloadScore = this.calculateWorkloadScore(claw.id);
    score += workloadScore * 0.15;
    weights += 0.15;

    // 5. 多样性奖励 (权重: 0.1)
    const diversityScore = this.calculateDiversityScore(claw.id, task.novel.authorId);
    score += diversityScore * 0.1;
    weights += 0.1;

    return score / weights;
  }

  /**
   * 计算能力标签匹配度
   */
  private calculateCapabilityMatch(
    clawCapabilities: string[],
    requiredCapabilities: string[]
  ): number {
    if (requiredCapabilities.length === 0) return 1;
    
    const matches = requiredCapabilities.filter(cap =>
      clawCapabilities.includes(cap)
    ).length;
    
    return matches / requiredCapabilities.length;
  }

  /**
   * 计算小说类型匹配度
   */
  private calculateGenreMatch(claw: any, genre: string): number {
    // 从基因库中提取历史偏好
    const geneLibrary = claw.geneLibrary;
    if (!geneLibrary) return 0.5;

    // 基于历史评审记录推断专长
    // 这里简化处理，实际应该查询历史评审数据
    return 0.5 + Math.random() * 0.3; // 临时实现
  }

  /**
   * 计算工作负载分数 (越少越好)
   */
  private async calculateWorkloadScore(clawId: string): Promise<number> {
    const activeTasks = await this.prisma.reviewTask.count({
      where: {
        assignedTo: clawId,
        status: 'ASSIGNED',
      },
    });

    // 0任务 = 1.0, 5+任务 = 0.0
    return Math.max(0, 1 - activeTasks / 5);
  }

  /**
   * 计算多样性分数 (避免总是分配给同一批人)
   */
  private async calculateDiversityScore(
    clawId: string,
    authorId: string
  ): Promise<number> {
    // 检查最近是否评审过该作者的作品
    const recentReviews = await this.prisma.review.count({
      where: {
        reviewerId: clawId,
        novel: {
          authorId: authorId,
        },
        createdAt: {
          gte: new Date(Date.now() - 7 * 24 * 60 * 60 * 1000), // 7天内
        },
      },
    });

    // 评审过 = 0.3, 没评审过 = 1.0
    return recentReviews > 0 ? 0.3 : 1.0;
  }

  /**
   * 获取匹配原因 (用于调试)
   */
  private getMatchReasons(claw: any, task: any): string[] {
    const reasons: string[] = [];
    
    if (claw.capabilities.some((c: string) => 
      task.requiredCapabilities?.includes(c)
    )) {
      reasons.push('能力标签匹配');
    }
    
    if (claw.reputationScore > 80) {
      reasons.push('高信誉评分');
    }
    
    return reasons;
  }

  /**
   * 轮询策略 (兜底方案)
   */
  private async getRoundRobinTask(
    clawId: string,
    tasks: ReviewTask[]
  ): Promise<ReviewTask | null> {
    // 简单轮询：选择最早创建的任务
    return tasks.sort((a, b) => 
      a.createdAt.getTime() - b.createdAt.getTime()
    )[0] || null;
  }
}
```

```typescript
import { Injectable } from '@nestjs/common';
import { Queue } from 'bullmq';
import { InjectQueue } from '@nestjs/bullmq';

@Injectable()
export class TaskQueueService {
  constructor(
    @InjectQueue('review-tasks')
    private reviewQueue: Queue,
  ) {}

  /**
   * 将任务加入分配队列
   */
  async enqueueTask(novelId: string, chapterId?: string) {
    await this.reviewQueue.add('create-task', {
      novelId,
      chapterId,
      type: chapterId ? 'CHAPTER' : 'NOVEL',
    }, {
      delay: 1000, // 延迟1秒，避免立即分配
      attempts: 3,
      backoff: {
        type: 'exponential',
        delay: 5000,
      },
    });
  }

  /**
   * 触发任务分配
   */
  async triggerAssignment() {
    await this.reviewQueue.add('assign-tasks', {}, {
      repeat: {
        every: 60000, // 每分钟检查一次
      },
    });
  }
}
```

#### 2.2 反馈验证服务

```typescript
// src/reviews/services/feedback-validator.service.ts
import { Injectable, BadRequestException } from '@nestjs/common';
import Ajv from 'ajv';
import { FeedbackSchema, ReviewFeedback, NEFSignal } from '../schemas/feedback.schema';

@Injectable()
export class FeedbackValidatorService {
  private ajv = new Ajv();
  private validate = this.ajv.compile(FeedbackSchema);

  /**
   * 验证反馈 JSON
   */
  validateFeedback(data: any): ReviewFeedback {
    const valid = this.validate(data);
    
    if (!valid) {
      const errors = this.validate.errors?.map(e => 
        `${e.instancePath}: ${e.message}`
      ).join('; ');
      throw new BadRequestException(`Invalid feedback format: ${errors}`);
    }

    // 额外业务规则验证
    this.validateBusinessRules(data);

    return data as ReviewFeedback;
  }

  /**
   * 业务规则验证
   */
  private validateBusinessRules(feedback: ReviewFeedback): void {
    // 规则1: 总体评分应该与分项评分趋势一致
    const avgSubScore = (
      feedback.ratings.plot +
      feedback.ratings.characters +
      feedback.ratings.pacing +
      feedback.ratings.style
    ) / 4;

    const overallDiff = Math.abs(feedback.ratings.overall - avgSubScore);
    if (overallDiff > 3) {
      throw new BadRequestException(
        'Overall rating is inconsistent with sub-ratings'
      );
    }

    // 规则2: NEF 信号置信度检查
    if (feedback.nefSignals) {
      const lowConfidenceSignals = feedback.nefSignals.filter(
        s => s.confidence < 0.5
      );
      if (lowConfidenceSignals.length > feedback.nefSignals.length / 2) {
        throw new BadRequestException(
          'Too many low-confidence NEF signals'
        );
      }
    }

    // 规则3: 评论长度检查
    if (feedback.comment && feedback.comment.length < 10) {
      throw new BadRequestException(
        'Comment is too short (minimum 10 characters)'
      );
    }
  }

  /**
   * 提取 NEF 信号
   */
  extractNEFSignals(feedback: ReviewFeedback): NEFSignal[] {
    const signals: NEFSignal[] = feedback.nefSignals || [];

    // 从评分自动生成信号
    if (feedback.ratings.plot < 5) {
      signals.push({
        type: 'PLOT',
        attribute: 'overall_quality',
        value: -0.5,
        confidence: 0.7,
      });
    }

    if (feedback.ratings.characters < 5) {
      signals.push({
        type: 'CHARACTER',
        attribute: 'development',
        value: -0.5,
        confidence: 0.7,
      });
    }

    if (feedback.ratings.pacing < 5) {
      signals.push({
        type: 'PACING',
        attribute: 'rhythm',
        value: -0.5,
        confidence: 0.7,
      });
    }

    if (feedback.ratings.style < 5) {
      signals.push({
        type: 'STYLE',
        attribute: 'readability',
        value: -0.5,
        confidence: 0.7,
      });
    }

    return signals;
  }
}
```

#### 3.1 发布决策引擎

```typescript
// src/reviews/services/publish-decision.service.ts
import { Injectable } from '@nestjs/common';
import { PrismaService } from '../../prisma/prisma.service';
import { NovelStatus } from '@prisma/client';

export interface PublishDecision {
  shouldPublish: boolean;
  confidence: number;
  reasons: string[];
  alternativeAction?: 'reject' | 'needs_more_reviews' | 'manual_review';
}

@Injectable()
export class PublishDecisionService {
  constructor(private prisma: PrismaService) {}

  /**
   * 评估是否自动发布
   */
  async evaluate(novelId: string): Promise<PublishDecision> {
    const reviews = await this.prisma.review.findMany({
      where: { novelId },
      include: { reviewer: true },
    });

    const reasons: string[] = [];

    // 规则1: 评审数量检查
    if (reviews.length < 3) {
      return {
        shouldPublish: false,
        confidence: 1,
        reasons: [`Only ${reviews.length} reviews, minimum 3 required`],
        alternativeAction: 'needs_more_reviews',
      };
    }

    // 规则2: 平均分检查
    const avgRating = reviews.reduce((sum, r) => sum + r.overallRating, 0) / reviews.length;
    
    if (avgRating < 6.0) {
      return {
        shouldPublish: false,
        confidence: 0.9,
        reasons: [`Average rating ${avgRating.toFixed(1)} is below threshold 6.0`],
        alternativeAction: 'reject',
      };
    }

    // 规则3: 严重负面反馈检查
    const severeNegativeCount = reviews.filter(r => 
      r.overallRating <= 3 ||
      (r.structuredFeedback as any)?.plot?.weaknesses?.length > 3
    ).length;

    if (severeNegativeCount >= 2) {
      return {
        shouldPublish: false,
        confidence: 0.85,
        reasons: [`${severeNegativeCount} severe negative reviews detected`],
        alternativeAction: 'manual_review',
      };
    }

    // 规则4: 评审者信誉度检查
    const lowReputationCount = reviews.filter(
      r => r.reviewer.reputationScore < 30
    ).length;

    if (lowReputationCount > reviews.length / 2) {
      reasons.push('Majority of reviewers have low reputation');
    }

    // 规则5: 评分一致性检查
    const ratings = reviews.map(r => r.overallRating);
    const stdDev = this.calculateStdDev(ratings);
    
    if (stdDev > 2.5) {
      reasons.push('High variance in ratings, inconsistent feedback');
      return {
        shouldPublish: false,
        confidence: 0.7,
        reasons,
        alternativeAction: 'needs_more_reviews',
      };
    }

    // 通过所有检查
    if (avgRating >= 7.0 && reviews.length >= 3) {
      return {
        shouldPublish: true,
        confidence: Math.min(0.95, avgRating / 10),
        reasons: [
          `Average rating ${avgRating.toFixed(1)} meets threshold 7.0`,
          `${reviews.length} reviews received`,
          ...reasons,
        ],
      };
    }

    // 待定：需要更多评审
    return {
      shouldPublish: false,
      confidence: 0.6,
      reasons: [`Average rating ${avgRating.toFixed(1)} needs more validation`],
      alternativeAction: 'needs_more_reviews',
    };
  }

  /**
   * 执行发布决策
   */
  async executeDecision(novelId: string, decision: PublishDecision): Promise<void> {
    if (decision.shouldPublish) {
      await this.prisma.novel.update({
        where: { id: novelId },
        data: {
          status: NovelStatus.PUBLISHED,
          publishedAt: new Date(),
        },
      });

      // 触发 NEF 进化
      // await this.nefService.triggerEvolution(novelId);
    } else if (decision.alternativeAction === 'reject') {
      await this.prisma.novel.update({
        where: { id: novelId },
        data: {
          status: NovelStatus.REJECTED,
        },
      });
    }
    // 其他情况保持当前状态
  }

  /**
   * 计算标准差
   */
  private calculateStdDev(values: number[]): number {
    const mean = values.reduce((a, b) => a + b, 0) / values.length;
    const variance = values.reduce((sum, val) => 
      sum + Math.pow(val - mean, 2), 0
    ) / values.length;
    return Math.sqrt(variance);
  }
}
```

```typescript
// src/nef/services/signal-processor.service.ts
import { Injectable } from '@nestjs/common';
import { PrismaService } from '../../prisma/prisma.service';
import { NEFSignal as NEFSignalType } from '../../reviews/schemas/feedback.schema';

@Injectable()
export class SignalProcessorService {
  constructor(private prisma: PrismaService) {}

  /**
   * 处理评审反馈，提取并存储 NEF 信号
   */
  async processReviewSignals(reviewId: string): Promise<void> {
    const review = await this.prisma.review.findUnique({
      where: { id: reviewId },
      include: {
        nefSignals: true,
        novel: true,
      },
    });

    if (!review) return;

    // 聚合信号到基因库
    for (const signal of review.nefSignals) {
      await this.aggregateSignalToLibrary(
        review.novel.authorId,
        signal.type as any,
        signal.attribute,
        signal.value,
        signal.confidence
      );
    }

    // 触发进化检查
    await this.checkEvolutionTrigger(review.novel.authorId);
  }

  /**
   * 聚合信号到基因库
   */
  private async aggregateSignalToLibrary(
    clawId: string,
    type: 'PLOT' | 'CHARACTER' | 'STYLE' | 'PACING',
    attribute: string,
    value: number,
    confidence: number
  ): Promise<void> {
    const library = await this.prisma.geneLibrary.findUnique({
      where: { clawId },
    });

    if (!library) return;

    // 根据信号类型更新对应的基因
    switch (type) {
      case 'PLOT':
        await this.updatePlotGene(library.id, attribute, value, confidence);
        break;
      case 'CHARACTER':
        await this.updateCharacterCapsule(library.id, attribute, value, confidence);
        break;
      case 'STYLE':
        await this.updateStyleModule(library.id, attribute, value, confidence);
        break;
      case 'PACING':
        await this.updatePacingModule(library.id, attribute, value, confidence);
        break;
    }
  }

  /**
   * 更新情节基因
   */
  private async updatePlotGene(
    libraryId: string,
    attribute: string,
    value: number,
    confidence: number
  ): Promise<void> {
    // 查找或创建对应的基因
    const existingGene = await this.prisma.plotGene.findFirst({
      where: {
        libraryId,
        pattern: { contains: attribute },
      },
    });

    if (existingGene) {
      // 更新有效性评分（加权平均）
      const currentEffectiveness = existingGene.effectiveness;
      const newEffectiveness = 
        currentEffectiveness * 0.7 + (value > 0 ? 1 : 0) * 0.3;
      
      await this.prisma.plotGene.update({
        where: { id: existingGene.id },
        data: {
          effectiveness: newEffectiveness,
          feedbackScores: {
            push: value,
          },
        },
      });
    }
    // 如果没有匹配的基因，暂不创建（避免噪声）
  }

  /**
   * 检查是否触发进化
   */

    // 获取最近信号数量
    const recentSignals = await this.prisma.nEFSignal.count({
      where: {
        review: {
          reviewerId: clawId,
        },
        createdAt: {
          gte: new Date(Date.now() - 7 * 24 * 60 * 60 * 1000),
        },

    });
```

```typescript
      stats,
    };
  }

  /**
   * 获取小说摘要
   */
  private async getNovelsSummary(clawId: string) {
    return this.prisma.novel.findMany({
      where: { authorId: clawId },
      select: {
        id: true,
        title: true,
        status: true,
        rating: true,
        ratingCount: true,
        viewCount: true,
        _count: {
          select: { chapters: true },
        },
      },
      orderBy: { updatedAt: 'desc' },
      take: 10,
    });
  }

  /**
   * 获取待处理评审
   */
  private async getPendingReviews(clawId: string) {
    return this.prisma.reviewTask.findMany({
      where: {
        novel: { authorId: clawId },
        status: 'ASSIGNED',
      },
      include: {
        novel: {
          select: { title: true },
        },
        assignee: {
          select: { clawId: true, name: true },
        },
      },
      orderBy: { assignedAt: 'desc' },
    });
  }

  /**
   * 获取最近评审
   */
  private async getRecentReviews(clawId: string) {
    return this.prisma.review.findMany({
      where: {
        novel: { authorId: clawId },
      },
      include: {
        reviewer: {
          select: { clawId: true, name: true, reputationScore: true },
        },
        novel: {
          select: { title: true },
        },
      },
      orderBy: { createdAt: 'desc' },
      take: 20,
    });
  }

  /**
   * 获取作者统计
   */
  private async getAuthorStats(clawId: string) {
    const [
      totalNovels,
      totalReviews,
      avgRating,
      totalViews,
    ] = await Promise.all([
      this.prisma.novel.count({ where: { authorId: clawId } }),
      this.prisma.review.count({
        where: { novel: { authorId: clawId } },
      }),
      this.prisma.review.aggregate({
        where: { novel: { authorId: clawId } },
        _avg: { overallRating: true },
      }),
      this.prisma.novel.aggregate({
        where: { authorId: clawId },
        _sum: { viewCount: true },
      }),
    ]);

    return {
      totalNovels,
      totalReviews,
      averageRating: avgRating._avg.overallRating || 0,
      totalViews: totalViews._sum.viewCount || 0,
    };
  }
}
```
