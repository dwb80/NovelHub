## 2. 详细设计

### 2.1 页面组件设计

**文件**: `apps/frontend/src/app/ranking/page.tsx`

```typescript
'use client';

import { useState } from 'react';
import { useQuery } from '@tanstack/react-query';
import { Header } from '@/components/layout/header';
import { Footer } from '@/components/layout/footer';
import { RankingTabs } from '@/components/ranking/ranking-tabs';
import { CategoryFilter } from '@/components/ranking/category-filter';
import { TopThreeSection } from '@/components/ranking/top-three-section';
import { RankingList } from '@/components/ranking/ranking-list';
import { TimePeriodSelector } from '@/components/ranking/time-period-selector';
import api from '@/lib/api';

type RankingType = 'popular' | 'rising' | 'new' | 'favorite';
type TimePeriod = 'daily' | 'weekly' | 'monthly' | 'all';

export default function RankingPage() {
  const [rankingType, setRankingType] = useState<RankingType>('popular');
  const [category, setCategory] = useState<string>('all');
  const [timePeriod, setTimePeriod] = useState<TimePeriod>('weekly');
  const [page, setPage] = useState(1);

  const { data, isLoading } = useQuery({
    queryKey: ['ranking', rankingType, category, timePeriod, page],
    queryFn: async () => {
      const response = await api.get('/ranking', {
        params: { 
          type: rankingType, 
          category, 
          period: timePeriod,
          page,
          limit: 20
        }
      });
      return response.data;
    },
    staleTime: 5 * 60 * 1000, // 5分钟缓存
  });

  // ...渲染逻辑
}
```

### 2.3 数据结构

```typescript
interface RankingItem {
  rank: number;
  novel: {
    id: string;
    title: string;
    authorName: string;
    coverImage: string;
    category: string;
    tags: string[];
  };
  stats: {
    totalViews: number;
    totalFavorites: number;
    totalComments: number;
    score: number;
  };
  trend: {
    direction: 'up' | 'down' | 'stable';
    change: number; // 排名变化
  };
  isNew: boolean; // 是否新上榜
}

interface RankingData {
  type: string;
  period: string;
  updatedAt: string;
  topThree: RankingItem[];
  list: RankingItem[];
  total: number;
  page: number;
  hasMore: boolean;
}
```

- 铜牌：尺寸 160x220px，铜色边框
- 位置：金牌居中偏下，银牌左偏上，铜牌右偏上

## 5. 缓存策略

### 5.1 Redis 缓存
- 排行榜数据缓存5分钟
- 使用 Sorted Set 存储排名
- Key 格式: `ranking:{type}:{category}:{period}`

### 5.2 前端缓存
- React Query 缓存5分钟
- 预加载相邻榜单数据
