```typescript
// 字数范围映射
const wordCountRangeMap = {
  'lt10w': { lt: 100000 },
  '10w30w': { gte: 100000, lt: 300000 },
  '30w50w': { gte: 300000, lt: 500000 },
  '50w100w': { gte: 500000, lt: 1000000 },
  'gt100w': { gte: 1000000 },
};

// Prisma where 条件构建
const where: any = {};

if (filters.targetAudience && filters.targetAudience !== 'all') {
  where.targetAudience = filters.targetAudience.toUpperCase();
}

if (filters.serialStatus && filters.serialStatus !== 'all') {
  where.serialStatus = filters.serialStatus.toUpperCase();
}

if (filters.wordCountRange && filters.wordCountRange !== 'all') {
  where.wordCount = wordCountRangeMap[filters.wordCountRange];
}
```

### 4.2 状态管理
```typescript
// 筛选状态
interface FilterState {
  targetAudience: string;  // 'all' | 'male' | 'female'
  serialStatus: string;    // 'all' | 'ongoing' | 'completed'
  wordCountRange: string;  // 'all' | 'lt10w' | '10w30w' | ...
  category: string;        // 'all' | 'XUANHUAN' | ...
}

// URL 同步
const updateUrlParams = (filters: FilterState, page: number) => {
  const params = new URLSearchParams();
  if (page > 1) params.set('page', page.toString());
  Object.entries(filters).forEach(([key, value]) => {
    if (value && value !== 'all') {
      params.set(key, value);
    }
  });
  router.push(queryString ? `?${queryString}` : '/novels');
};
```

## 5. 缓存策略

### 5.1 缓存键生成
```typescript
private generateListCacheKey(
  page: number,
  limit: number,
  filters?: { ... }
): string {
  const filterKey = filters 
    ? Object.entries(filters)
        .filter(([_, v]) => v && v !== 'all')
        .map(([k, v]) => `${k}:${v}`)
        .join(':')
    : 'all';
  return `novels:list:${page}:${limit}:${filterKey}`;
}
```

### 5.2 缓存失效
- 筛选条件变化时，自动生成新的缓存键
- 小说数据更新时，清除相关缓存
- authorId 查询不使用缓存（保证数据一致性）

## 6. 性能优化

### 6.1 数据库优化
- 为筛选字段添加索引
- 使用 Prisma 的 select 只查询需要的字段
- 分页查询避免大数据量加载

### 6.2 前端优化
- 使用 React Query 缓存数据
- 图片懒加载
- 分页组件避免一次性渲染过多页码
