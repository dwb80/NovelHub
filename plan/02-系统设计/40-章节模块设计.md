## 2. 详细设计

### 2.1 章节列表页面

**文件**: `apps/frontend/src/app/novels/[id]/chapters/page.tsx`

```typescript
import { Metadata } from 'next';
import { notFound } from 'next/navigation';
import { Header } from '@/components/layout/header';
import { Footer } from '@/components/layout/footer';
import { Breadcrumb } from '@/components/ui/breadcrumb';
import { ChapterList } from '@/components/chapters/chapter-list';
import { NovelInfoHeader } from '@/components/chapters/novel-info-header';
import { ChapterToolbar } from '@/components/chapters/chapter-toolbar';
import api from '@/lib/api';

interface PageProps {
  params: { id: string };
  searchParams: { page?: string; order?: 'asc' | 'desc' };
}

export async function generateMetadata({ params }: PageProps): Promise<Metadata> {
  const novel = await api.get(`/novels/${params.id}`).catch(() => null);
  if (!novel) return { title: '章节列表 - NovelHub' };
  return {
    title: `${novel.data.title} - 章节目录 - NovelHub`,
    description: `${novel.data.title}共${novel.data.totalChapters}章`,
  };
}

export default async function ChaptersPage({ params, searchParams }: PageProps) {
  const page = parseInt(searchParams.page || '1');
  const order = searchParams.order || 'asc';
  
  const [novel, chapters] = await Promise.all([
    api.get(`/novels/${params.id}`).catch(() => null),
    api.get(`/novels/${params.id}/chapters`, {
      params: { page, limit: 50, order }
    }).catch(() => null),
  ]);

  if (!novel || !chapters) notFound();

  return (
    <div className="min-h-screen flex flex-col">
      <Header />
      <main className="flex-1 container mx-auto px-4 py-8">
        <Breadcrumb items={[...]} />
        <NovelInfoHeader novel={novel.data} />
        <ChapterToolbar 
          totalChapters={novel.data.totalChapters}
          currentOrder={order}
        />
        <ChapterList 
          chapters={chapters.data.items}
          novelId={params.id}
          currentPage={page}
          totalPages={chapters.data.totalPages}
        />
      </main>
      <Footer />
    </div>
  );
}
```

### 2.2 章节阅读页面

**文件**: `apps/frontend/src/app/novels/[id]/chapters/[chapterId]/page.tsx`

```typescript
'use client';

import { useEffect, useState } from 'react';
import { useQuery, useMutation } from '@tanstack/react-query';
import { useParams, useRouter } from 'next/navigation';
import { ReaderToolbar } from '@/components/reader/reader-toolbar';
import { ChapterContent } from '@/components/reader/chapter-content';
import { ReaderNav } from '@/components/reader/reader-nav';
import { SettingsPanel } from '@/components/reader/settings-panel';
import { useReaderSettings } from '@/hooks/use-reader-settings';
import { useReadingProgress } from '@/hooks/use-reading-progress';
import api from '@/lib/api';

export default function ChapterReaderPage() {
  const { id: novelId, chapterId } = useParams();
  const router = useRouter();
  const [showSettings, setShowSettings] = useState(false);
  const { settings, updateSettings } = useReaderSettings();
  const { updateProgress } = useReadingProgress(novelId as string);

  const { data: chapter, isLoading } = useQuery({
    queryKey: ['chapter', chapterId],
    queryFn: () => api.get(`/chapters/${chapterId}`).then(r => r.data),
  });

  const { data: navigation } = useQuery({
    queryKey: ['chapter-nav', novelId, chapterId],
    queryFn: () => api.get(`/novels/${novelId}/chapters/navigation`, {
      params: { currentChapterId: chapterId }
    }).then(r => r.data),
  });

  // 自动保存阅读进度
  useEffect(() => {
    if (chapter) {
      updateProgress(chapterId as string, 100);
    }
  }, [chapter, chapterId, updateProgress]);

  // ...渲染逻辑
}
```

## 3. API 接口设计

### 3.1 获取章节列表
```http
GET /api/v1/novels/{novelId}/chapters?page={page}&limit={limit}&order={order}
```

**响应格式**:
```json
{
  "items": [
    {
      "id": "uuid",
      "number": 1,
      "title": "第一章 穿越",
      "wordCount": 3256,
      "publishedAt": "2024-01-01T00:00:00Z",
      "isLocked": false
    }
  ],
  "total": 100,
  "page": 1,
  "totalPages": 2
}
```

### 3.2 获取章节内容
```http
GET /api/v1/chapters/{chapterId}
```

### 3.3 获取章节导航
```http
GET /api/v1/novels/{novelId}/chapters/navigation?currentChapterId={id}
```

### 3.4 更新阅读进度
```http
PUT /api/v1/reading-progress
{
  "novelId": "uuid",
  "chapterId": "uuid",
  "progress": 100
}
```
