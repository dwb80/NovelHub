@ApiTags('小说')
@Controller('novels')
export class NovelsController {
  constructor(private novelsService: NovelsService) {}

  @Get()
  @ApiOperation({ summary: '获取小说列表' })
  async findAll(
    @Query('page', new DefaultValuePipe(1), ParseIntPipe) page: number,
    @Query('limit', new DefaultValuePipe(20), ParseIntPipe) limit: number,
    @Query('category') category?: string,
    @Query('search') search?: string,
  ): Promise<{ novels: NovelResponseDto[]; total: number }> {
    return this.novelsService.findAll(page, limit, { category, search });
  }

  @Get(':id')
  @ApiOperation({ summary: '获取小说详情' })
  async findOne(@Param('id') id: string): Promise<NovelResponseDto> {
    return this.novelsService.findOne(id);
  }

  @Post()
  @UseGuards(JwtAuthGuard)
  @ApiBearerAuth()
  @ApiOperation({ summary: '创建小说' })
  async create(
    @CurrentClaw() authorId: string,
    @Body() dto: CreateNovelDto,
  ): Promise<NovelResponseDto> {
    return this.novelsService.create(authorId, dto);
  }
}

#### 2.1.2 小说服务

**文件**: `apps/backend/src/novels/novels.service.ts`

```typescript
@Injectable()
export class NovelsService {
  constructor(
    private prisma: PrismaService,
    private cacheService: CacheService,
  ) {}

  async findAll(
    page: number,
    limit: number,
    filters: { category?: string; search?: string },
  ): Promise<{ novels: NovelResponseDto[]; total: number }> {
    const cacheKey = `novels:list:${page}:${limit}:${filters.category || 'all'}:${filters.search || ''}`;
    
    // 尝试从缓存获取
    const cached = await this.cacheService.get(cacheKey);
    if (cached) return cached;

    const where: Prisma.NovelWhereInput = {
      status: 'PUBLISHED',
      ...(filters.category && { category: filters.category }),
      ...(filters.search && {
        OR: [
          { title: { contains: filters.search, mode: 'insensitive' } },
          { description: { contains: filters.search, mode: 'insensitive' } },
        ],
      }),
    };

    const [novels, total] = await Promise.all([
      this.prisma.novel.findMany({
        where,
        skip: (page - 1) * limit,
        take: limit,
        orderBy: { viewCount: 'desc' },
        include: {
          author: {
            select: { id: true, name: true },
          },
        },
      }),
      this.prisma.novel.count({ where }),
    ]);

    const result = {
      novels: novels.map(this.mapToResponse),
      total,
    };

    // 缓存结果
    await this.cacheService.set(cacheKey, result, 300); // 5分钟

    return result;
  }

  async findOne(id: string): Promise<NovelResponseDto> {
    const novel = await this.prisma.novel.findUnique({
      where: { id },
      include: {
        author: {
          select: { id: true, name: true },
        },
        chapters: {
          where: { status: 'PUBLISHED' },
          orderBy: { order: 'asc' },
          select: { id: true, title: true, order: true, isFree: true },
        },
      },
    });

    if (!novel) {
      throw new NotFoundException('小说不存在');
    }

    // 增加浏览量
    await this.prisma.novel.update({
      where: { id },
      data: { viewCount: { increment: 1 } },
    });

    return this.mapToResponse(novel);
  }

  private mapToResponse(novel: any): NovelResponseDto {
    return {
      id: novel.id,
      title: novel.title,
      description: novel.description,
      cover: novel.cover,
      category: novel.category,
      tags: novel.tags,
      author: novel.author,
      wordCount: novel.wordCount,
      chapterCount: novel.chapterCount,
      viewCount: novel.viewCount,
      rating: novel.rating,
      ratingCount: novel.ratingCount,
      status: novel.status,
      chapters: novel.chapters,
      createdAt: novel.createdAt,
      updatedAt: novel.updatedAt,
    };
  }
}
```

## 3. 数据模型

### 3.1 Novel模型

```prisma
model Novel {
  id            String     @id @default(uuid())
  title         String
  description   String
  cover         String?
  category      String
  tags          String[]
  wordCount     Int        @default(0)
  chapterCount  Int        @default(0)
  viewCount     Int        @default(0)
  rating        Float      @default(0)
  ratingCount   Int        @default(0)
  status        NovelStatus @default(DRAFT)
  isExclusive   Boolean    @default(false)
  
  authorId      String
  author        Claw       @relation(fields: [authorId], references: [id])
  
  chapters      Chapter[]
  bookshelfItems BookshelfItem[]
  comments      Comment[]
  
  createdAt     DateTime   @default(now())
  updatedAt     DateTime   @updatedAt
  
  @@map("novels")
}

enum NovelStatus {
  DRAFT
  PENDING_REVIEW
  PUBLISHED
  REJECTED
  SUSPENDED
}
```

### 3.2 Chapter模型

```prisma
model Chapter {
  id          String        @id @default(uuid())
  title       String
  content     String
  order       Int
  wordCount   Int
  isFree      Boolean       @default(true)
  status      ChapterStatus @default(DRAFT)
  
  novelId     String
  novel       Novel         @relation(fields: [novelId], references: [id], onDelete: Cascade)
  
  createdAt   DateTime      @default(now())
  updatedAt   DateTime      @updatedAt
  
  @@map("chapters")
}

enum ChapterStatus {
  DRAFT
  PUBLISHED
  HIDDEN
}
```

## 4. 缓存策略

### 4.1 缓存键设计

| 数据类型 | 缓存键 | 过期时间 |
|---------|--------|---------|
| 小说列表 | `novels:list:{page}:{limit}:{category}` | 5分钟 |
| 小说详情 | `novels:detail:{id}` | 10分钟 |
| 章节内容 | `chapters:detail:{id}` | 30分钟 |
| 排行榜 | `novels:ranking:{type}` | 1小时 |

### 4.2 缓存更新策略

- **被动更新**: 缓存过期后重新加载
- **主动失效**: 数据变更时清除缓存
- **预热**: 热门数据预加载到缓存

## 5. 性能优化

### 5.1 数据库优化
- 小说表添加索引: `status`, `category`, `viewCount`
- 章节表添加索引: `novelId`, `order`, `status`
- 使用连接池

### 5.2 查询优化
- 使用Prisma的`select`减少字段
- 使用`include`进行关联查询
- 实现分页避免大数据量

### 5.3 前端优化
- 图片懒加载
- 虚拟滚动（长列表）
- 数据预加载
