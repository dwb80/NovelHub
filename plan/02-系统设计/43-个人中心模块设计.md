## 2. 详细设计

### 2.1 个人中心页面

**文件**: `apps/frontend/src/app/profile/page.tsx`

```typescript
'use client';

import { useQuery } from '@tanstack/react-query';
import { Header } from '@/components/layout/header';
import { Footer } from '@/components/layout/footer';
import { ProfileCard } from '@/components/profile/profile-card';
import { ReadingStats } from '@/components/profile/reading-stats';
import { AchievementGrid } from '@/components/profile/achievement-grid';
import { QuickLinks } from '@/components/profile/quick-links';
import { useAuth } from '@/hooks/use-auth';
import api from '@/lib/api';

export default function ProfilePage() {
  const { user } = useAuth();

  const { data: profile } = useQuery({
    queryKey: ['profile'],
    queryFn: () => api.get('/profile').then(r => r.data),
  });

  const { data: stats } = useQuery({
    queryKey: ['reading-stats'],
    queryFn: () => api.get('/profile/reading-stats').then(r => r.data),
  });

  const { data: achievements } = useQuery({
    queryKey: ['achievements'],
    queryFn: () => api.get('/profile/achievements').then(r => r.data),
  });

  return (
    <div className="min-h-screen flex flex-col">
      <Header />
      <main className="flex-1 container mx-auto px-4 py-8">
        <ProfileCard profile={profile} />
        <ReadingStats stats={stats} />
        <AchievementGrid achievements={achievements} />
        <QuickLinks />
      </main>
      <Footer />
    </div>
  );
}
```

### 2.3 数据结构

```typescript
interface Profile {
  id: string;
  displayName: string;
  avatar: string;
  bio: string;
  level: number;
  experience: number;
  nextLevelExperience: number;
  joinedAt: string;
}

interface ReadingStats {
  totalDays: number;
  totalWords: number;
  totalHours: number;
  categoryDistribution: {
    category: string;
    percentage: number;
  }[];
  weeklyTrend: {
    date: string;
    words: number;
  }[];
}

interface Achievement {
  id: string;
  name: string;
  description: string;
  icon: string;
  isUnlocked: boolean;
  unlockedAt?: string;
  progress: number;
  maxProgress: number;
}
```

## 3. API 接口设计

### 3.1 获取个人信息
```http
GET /api/v1/profile
```

**响应格式**:
```json
{
  "id": "uuid",
  "displayName": "用户名",
  "avatar": "https://...",
  "bio": "个人简介",
  "level": 5,
  "experience": 4500,
  "nextLevelExperience": 5000,
  "joinedAt": "2024-01-01T00:00:00Z"
}
```

### 3.2 更新个人信息
```http
PUT /api/v1/profile
{
  "displayName": "新用户名",
  "bio": "新简介"
}
```

### 3.3 上传头像
```http
POST /api/v1/profile/avatar
Content-Type: multipart/form-data
```

### 3.4 获取阅读统计
```http
GET /api/v1/profile/reading-stats
```

### 3.5 获取成就列表
```http
GET /api/v1/profile/achievements
```

## 4. 等级系统设计

### 4.1 等级规则

| 等级 | 名称 | 所需经验值 | 特权 |
|------|------|-----------|------|
| 1 | 书虫 | 0 | 基础功能 |
| 2 | 书迷 | 500 | 评论高亮 |
| 3 | 书痴 | 1500 | 专属徽章 |
| 4 | 书狂 | 3000 | 优先客服 |
| 5 | 书圣 | 5000 | 认证标识 |

### 4.2 经验值获取

| 行为 | 经验值 |
|------|--------|
| 每日登录 | +10 |
| 阅读1章 | +5 |
| 发表评论 | +10 |
| 被点赞 | +2 |
| 收藏小说 | +5 |
| 完善资料 | +50 |
