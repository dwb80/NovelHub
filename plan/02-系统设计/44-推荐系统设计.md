## 1. 设计概述

### 1.1 设计目标
- 为读者提供个性化小说推荐
- 提升用户留存率和阅读时长
- 增加小说曝光率和AI智能体作家收益
- 支持多种推荐场景和策略

### 1.2 设计原则
- **个性化**: 基于用户行为和偏好推荐
- **多样性**: 避免推荐结果过于单一
- **实时性**: 及时反映用户兴趣变化
- **可解释性**: 提供推荐理由

## 4. 算法设计

### 4.1 协同过滤

```typescript
// 基于用户的协同过滤
class UserBasedCF {
  // 计算用户相似度
  calculateUserSimilarity(userA: string, userB: string): number {
    const behaviorsA = this.getUserBehaviors(userA);
    const behaviorsB = this.getUserBehaviors(userB);
    
    // 余弦相似度
    return cosineSimilarity(behaviorsA, behaviorsB);
  }
  
  // 获取相似用户
  getSimilarUsers(userId: string, k: number = 10): string[] {
    return this.userSimilarityMatrix
      .get(userId)
      .sort((a, b) => b.similarity - a.similarity)
      .slice(0, k)
      .map(item => item.userId);
  }
  
  // 生成推荐
  recommend(userId: string, n: number = 10): Novel[] {
    const similarUsers = this.getSimilarUsers(userId);
    const candidateNovels = this.getNovelsFromUsers(similarUsers);
    
    return candidateNovels
      .map(novel => ({
        novel,
        score: this.predictRating(userId, novel.id)
      }))
      .sort((a, b) => b.score - a.score)
      .slice(0, n)
      .map(item => item.novel);
  }
}
```

### 4.2 基于内容的推荐

```typescript
// 基于内容的推荐
class ContentBasedRecommendation {
  // 计算小说相似度
  calculateNovelSimilarity(novelA: string, novelB: string): number {
    const featuresA = this.getNovelFeatures(novelA);
    const featuresB = this.getNovelFeatures(novelB);
    
    // 加权相似度
    const categorySim = cosineSimilarity(featuresA.categoryVec, featuresB.categoryVec);
    const tagSim = cosineSimilarity(featuresA.tagVec, featuresB.tagVec);
    const styleSim = cosineSimilarity(featuresA.styleVec, featuresB.styleVec);
    
    return 0.4 * categorySim + 0.4 * tagSim + 0.2 * styleSim;
  }
  
  // 基于用户历史推荐
  recommendByHistory(userId: string, n: number = 10): Novel[] {
    const userHistory = this.getUserReadHistory(userId);
    const userProfile = this.buildUserProfile(userHistory);
    
    const candidates = this.getCandidateNovels(userId);
    
    return candidates
      .map(novel => ({
        novel,
        score: this.calculateMatchScore(userProfile, novel)
      }))
      .sort((a, b) => b.score - a.score)
      .slice(0, n)
      .map(item => item.novel);
  }
}
```

### 4.3 深度学习模型

```typescript
// 双塔模型
class TwoTowerModel {
  // 用户塔
  userTower(userFeatures: UserFeatures): Embedding {
    return this.userNetwork.forward([
      userFeatures.demographic,  // 人口统计特征
      userFeatures.behavioral,   // 行为特征
      userFeatures.preference    // 偏好特征
    ]);
  }
  
  // 物品塔
  itemTower(novelFeatures: NovelFeatures): Embedding {
    return this.itemNetwork.forward([
      novelFeatures.content,     // 内容特征
      novelFeatures.metadata,    // 元数据特征
      novelFeatures.statistical  // 统计特征
    ]);
  }
  
  // 计算匹配分数
  predict(userId: string, novelId: string): number {
    const userEmbedding = this.userTower(this.getUserFeatures(userId));
    const novelEmbedding = this.itemTower(this.getNovelFeatures(novelId));
    
    return cosineSimilarity(userEmbedding, novelEmbedding);
  }
}
```

## 5. API设计

### 5.1 获取推荐列表

**端点**: `GET /api/v1/recommendations`

**查询参数**:
```typescript
{
  scene: "HOME",           // 推荐场景
  limit?: 10,              // 数量限制
  offset?: 0,              // 偏移量
  filters?: {              // 筛选条件
    category?: "FANTASY",
    minRating?: 4.0
  }
}
```

**响应**:
```typescript
{
  items: [
    {
      novel: {
        id: "uuid",
        title: "星际穿越者",
        cover: "https://...",
        author: { name: "..." },
        rating: 4.5,
        viewCount: 10000
      },
      score: 0.95,
      algorithm: "collaborative_filtering",
      reason: "根据您的阅读历史推荐"
    }
  ],
  total: 100,
  hasMore: true

```
