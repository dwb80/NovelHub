## 2. 详细设计

### 2.1 分类页面组件

**文件**: `apps/frontend/src/app/category/page.tsx`

```typescript
import { Metadata } from 'next';
import { Header } from '@/components/layout/header';
import { Footer } from '@/components/layout/footer';
import { CategoryGrid } from '@/components/category/category-grid';
import { TagCloud } from '@/components/category/tag-cloud';
import api from '@/lib/api';

export const metadata: Metadata = {
  title: '小说分类 - NovelHub',
  description: '按分类浏览精彩小说',
};

export default async function CategoryPage() {
  const categories = await api.get('/categories');
  const hotTags = await api.get('/tags/hot');

  return (
    <div className="min-h-screen flex flex-col">
      <Header />
      <main className="flex-1 container mx-auto px-4 py-8">
        <h1 className="text-3xl font-bold mb-8">小说分类</h1>
        <CategoryGrid categories={categories.data} />
        <TagCloud tags={hotTags.data} />
      </main>
      <Footer />
    </div>
  );
}
```

```typescript
'use client';

export default function CategoryDetailPage() {
  const { slug } = useParams();
  const [filters, setFilters] = useState({
    status: 'all',
    wordCount: 'all',
    rating: 'all',
    sort: 'popular'
  });
  const [page, setPage] = useState(1);

  const { data: category } = useQuery({
    queryKey: ['category', slug],
    queryFn: () => api.get(`/categories/${slug}`).then(r => r.data),

  const { data: novels } = useQuery({
    queryKey: ['category-novels', slug, filters, page],
    queryFn: () => api.get(`/categories/${slug}/novels`, {
      params: { ...filters, page, limit: 20 }
    }).then(r => r.data),
  });

  // ...渲染逻辑
}
```

### 2.4 数据结构

```typescript
interface Category {
  id: string;
  slug: string;
  name: string;
  description: string;
  icon: string;
  novelCount: number;
  subCategories?: Category[];
}

interface Tag {
  id: string;
  name: string;
  slug: string;
  novelCount: number;
  hotness: number; // 热度值，用于标签大小
}

interface CategoryFilter {
  status: 'all' | 'ongoing' | 'completed';
  wordCount: 'all' | 'short' | 'medium' | 'long';
  rating: 'all' | 'high' | 'medium';
  sort: 'popular' | 'newest' | 'rating' | 'updated';
}
```

## 3. API 接口设计

### 3.1 获取分类列表
```http
GET /api/v1/categories
```

**响应格式**:
```json
{
  "categories": [
    {
      "id": "uuid",
      "slug": "xuanhuan",
      "name": "玄幻",
      "description": "修炼成仙，逆天改命",
      "icon": "https://...",
      "novelCount": 1256,
      "subCategories": [
        { "id": "uuid", "slug": "xiuzhen", "name": "修真", "novelCount": 456 }
      ]
    }
  ]
}
```

### 3.2 获取热门标签
```http
GET /api/v1/tags/hot?limit={limit}
```

### 3.3 获取分类下小说
```http
GET /api/v1/categories/{slug}/novels?status={status}&wordCount={wordCount}&rating={rating}&sort={sort}&page={page}
```

### 3.4 按标签搜索
```http
GET /api/v1/tags/{slug}/novels?page={page}
```

## 4. 标签云设计

### 4.1 视觉规格
- 标签大小根据热度动态调整
- 颜色根据分类区分
- 悬停时放大并显示小说数量

```typescript
const getTagSize = (hotness: number): string => {
  if (hotness > 10000) return 'text-2xl font-bold';
  if (hotness > 5000) return 'text-xl font-semibold';
  if (hotness > 1000) return 'text-lg';
  return 'text-base';
};
```
