## 1. 设计概述

### 1.1 设计目标
- 提升系统响应速度，降低数据库压力
- 支持高并发场景下的数据读取
- 提供分布式缓存一致性保障
- 实现多级缓存架构

### 1.2 设计原则
- **缓存穿透防护**: 防止查询不存在的数据
- **缓存击穿防护**: 防止热点数据失效
- **缓存雪崩防护**: 防止大量缓存同时失效
- **最终一致性**: 允许短暂的数据不一致

## 3. Redis缓存设计

### 3.1 缓存Key命名规范

```
格式: {namespace}:{entity}:{identifier}:{field}

示例:
- novel:detail:{novelId}           # 小说详情
- novel:chapter:{chapterId}        # 章节内容
- user:profile:{userId}            # 用户信息
- user:session:{sessionId}         # 会话信息
- ranking:hot:{date}               # 热门排行榜
- config:system:{key}              # 系统配置
```

### 3.2 缓存数据结构

#### 3.2.1 小说详情缓存 (Hash)

```redis
# Key: novel:detail:{novelId}
HSET novel:detail:123 \
  id "123" \
  title "星际穿越者" \
  authorId "456" \
  authorName "科幻作家" \
  category "SCI_FI" \
  tags "["科幻","穿越","冒险"]" \
  rating "4.5" \
  viewCount "10000" \
  chapterCount "50" \
  wordCount "150000" \
  status "ONGOING" \
  cover "https://cdn.example.com/cover.jpg" \
  description "一个关于星际穿越的故事..." \
  updatedAt "2026-04-19T10:00:00Z"

# TTL: 10分钟
EXPIRE novel:detail:123 600
```

#### 3.2.2 排行榜缓存 (Sorted Set)

```redis
# Key: ranking:hot:2026-04-19
ZADD ranking:hot:2026-04-19 \
  10000 "novel:123" \
  8500 "novel:456" \
  7200 "novel:789"

# 获取前10名
ZREVRANGE ranking:hot:2026-04-19 0 9 WITHSCORES

# TTL: 1小时
EXPIRE ranking:hot:2026-04-19 3600
```

#### 3.2.3 用户会话缓存 (String)

```redis
# Key: user:session:{sessionId}
SET user:session:abc123 \
  "{\"userId\":\"456\",\"roles\":[\"READER\"],\"exp\":1713520800}"

# TTL: 24小时
EXPIRE user:session:abc123 86400
```

#### 3.2.4 防重复提交 (String)

```redis
# Key: lock:submit:{userId}:{action}
SET lock:submit:456:comment "1" NX EX 5

# 检查是否存在
EXISTS lock:submit:456:comment
```

## 4. 缓存服务设计

### 4.1 缓存服务接口

```typescript
interface CacheService {
  // 基础操作
  get<T>(key: string): Promise<T | null>;
  set<T>(key: string, value: T, ttl?: number): Promise<void>;
```

### 4.2 多级缓存实现

```typescript
class MultiLevelCacheService implements CacheService {
  private localCache: NodeCache;      // L1: 本地缓存
  private redis: Redis;               // L2: Redis缓存
  
  constructor() {
    this.localCache = new NodeCache({
      stdTTL: 60,           // 默认1分钟
      checkperiod: 120,     // 检查周期2分钟
      maxKeys: 10000        // 最大键数
    });
    
    this.redis = new Redis({
      host: process.env.REDIS_HOST,
      port: parseInt(process.env.REDIS_PORT || '6379'),
      password: process.env.REDIS_PASSWORD,
      db: 0
    });
  }
  
  async get<T>(key: string): Promise<T | null> {
    // 1. 先查本地缓存
    const localValue = this.localCache.get<T>(key);
    if (localValue !== undefined) {
      return localValue;
    }
    
    // 2. 再查Redis
    const redisValue = await this.redis.get(key);
    if (redisValue) {
      const parsed = JSON.parse(redisValue);
      // 回填本地缓存
      this.localCache.set(key, parsed);
      return parsed;
    }
    
    return null;
  }
  
  async set<T>(
    key: string, 
    value: T, 
    ttl: number = 300
  ): Promise<void> {
    const serialized = JSON.stringify(value);
    
    // 1. 写入Redis
    await this.redis.setex(key, ttl, serialized);
    
    // 2. 写入本地缓存 (TTL设为Redis的1/5)
    this.localCache.set(key, value, Math.floor(ttl / 5));
  }
  
  async delete(key: string): Promise<void> {
    // 1. 删除本地缓存
    this.localCache.del(key);
    
    // 2. 删除Redis
    await this.redis.del(key);
  }
}
```

## 5. 缓存模式

### 5.1 Cache-Aside (旁路缓存)

```typescript
class CacheAsideService {
  async getNovelDetail(novelId: string): Promise<Novel | null> {
    const cacheKey = `novel:detail:${novelId}`;
    
    // 1. 查缓存
    const cached = await this.cache.get<Novel>(cacheKey);
    if (cached) {
      return cached;
    }
    
    // 2. 查数据库
    const novel = await this.prisma.novel.findUnique({
      where: { id: novelId }
    });
    
    // 3. 写入缓存
    if (novel) {
      await this.cache.set(cacheKey, novel, 600);
    }
    
    return novel;
  }
  
  async updateNovel(
    novelId: string, 
    data: UpdateNovelDto
  ): Promise<Novel> {
    // 1. 更新数据库
    const updated = await this.prisma.novel.update({
      where: { id: novelId },
      data
    });
    
    // 2. 删除缓存 (下次读取时重建)
    await this.cache.delete(`novel:detail:${novelId}`);
    
    return updated;
  }
}
```

### 5.2 Read-Through (直读缓存)

```typescript
class ReadThroughCache {
  async get<T>(
    key: string,
    loader: () => Promise<T>,
    ttl: number = 300
  ): Promise<T> {
    // 1. 查缓存
    const cached = await this.cache.get<T>(key);
    if (cached !== null) {
      return cached;
    }
    
    // 2. 获取分布式锁
    const lockKey = `lock:${key}`;
    const acquired = await this.cache.lock(lockKey, 10);
    
    if (!acquired) {
      // 等待后重试
      await sleep(100);
      return this.get(key, loader, ttl);
    }
    
    try {
      // 3. 双重检查
      const doubleCheck = await this.cache.get<T>(key);
      if (doubleCheck !== null) {
        return doubleCheck;
      }
      
      // 4. 加载数据
      const data = await loader();
      
      // 5. 写入缓存
      await this.cache.set(key, data, ttl);
      
      return data;
    } finally {
      await this.cache.unlock(lockKey);
    }
  }
}
```

### 5.3 Write-Through (直写缓存)

```typescript
class WriteThroughCache {
```

### 5.4 Write-Behind (异步写)

```typescript
class WriteBehindCache {
  private writeQueue: Queue;
  
  async set<T>(
    key: string,
    value: T,
    ttl: number = 300
  ): Promise<void> {
    // 1. 只写缓存
    await this.cache.set(key, value, ttl);
    
    // 2. 加入异步写队列
    await this.writeQueue.add('persist', {
      key,
      value,
      timestamp: Date.now()
    });
  }
  
  // 队列处理器
  async processWriteQueue(job: Job) {
    const { key, value } = job.data;
    await this.persistToDatabase(key, value);
  }
}
```

## 6. 缓存问题解决方案

### 6.1 缓存穿透

**问题**: 查询不存在的数据，每次都打到数据库

**解决方案**:

```typescript
class CachePenetrationProtection {
  async get<T>(key: string, loader: () => Promise<T | null>): Promise<T | null> {
    // 1. 查缓存
    const cached = await this.cache.get<T>(key);
    if (cached !== null) {
      return cached;
    }
    
    // 2. 检查空值缓存 (防止穿透)
    const nullKey = `null:${key}`;
    if (await this.cache.exists(nullKey)) {
      return null;
    }
    
    // 3. 查数据库
    const data = await loader();
    
    if (data) {
      // 4. 写入正常缓存
      await this.cache.set(key, data, 300);
    } else {
      // 5. 写入空值缓存 (短时间)
      await this.cache.set(nullKey, '1', 60);
    }
    
    return data;
  }
}
```

### 6.2 缓存击穿

**问题**: 热点数据过期瞬间，大量请求打到数据库

**解决方案**:

```typescript
class CacheBreakdownProtection {
  async get<T>(
    key: string,
    loader: () => Promise<T>,
    ttl: number = 300
  ): Promise<T> {
    const cached = await this.cache.get<T>(key);
    if (cached !== null) {
      return cached;
    }
    
    // 1. 获取分布式锁
    const lockKey = `lock:${key}`;
    const lock = await this.redlock.acquire([lockKey], 1000);
    
    try {
      // 2. 双重检查
      const doubleCheck = await this.cache.get<T>(key);
      if (doubleCheck !== null) {
        return doubleCheck;
      }
      
      // 3. 加载数据
      const data = await loader();
      
      // 4. 写入缓存
      await this.cache.set(key, data, ttl);
      
      return data;
    } finally {
      await lock.release();
    }
  }
}
```

### 6.3 缓存雪崩

**问题**: 大量缓存同时过期，数据库压力激增

**解决方案**:

```typescript
class CacheAvalancheProtection {
  async set<T>(
    key: string,
    value: T,
    baseTtl: number = 300
  ): Promise<void> {
    // 1. 添加随机偏移量 (±10%)
    const jitter = Math.floor(baseTtl * 0.1 * (Math.random() * 2 - 1));
    const ttl = baseTtl + jitter;
    
    await this.cache.set(key, value, ttl);
  }
  
  // 2. 热点数据永不过期 + 后台刷新
  async setHotData<T>(
    key: string,
    loader: () => Promise<T>
  ): Promise<void> {
    // 写入缓存，不设置过期时间
    const data = await loader();
    await this.cache.set(key, data);
    
    // 启动后台定时刷新
    this.scheduleRefresh(key, loader, 300);
  }
}
```

### 8.2 缓存分析

```typescript
// 缓存统计分析
class CacheAnalytics {
  async analyzeCacheEfficiency() {
    const info = await this.redis.info('stats');
    
    // 计算命中率
    const keyspaceHits = parseInt(info.keyspace_hits);
    const keyspaceMisses = parseInt(info.keyspace_misses);
    const hitRate = keyspaceHits / (keyspaceHits + keyspaceMisses);
    
    // 分析热点key
    const hotKeys = await this.redis.sendCommand([
      'SLOWLOG', 'GET', '10'
    ]);
    
    return {
      hitRate,
      hotKeys,
      memoryUsage: info.used_memory_human,
      totalKeys: info.db0?.keys || 0
    };
  }
}
```
