## 2. 详细设计

### 2.1 通知中心页面

**文件**: `apps/frontend/src/app/notifications/page.tsx`

```typescript
'use client';

import { useState } from 'react';
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
import { Header } from '@/components/layout/header';
import { Footer } from '@/components/layout/footer';
import { NotificationList } from '@/components/notifications/notification-list';
import { NotificationFilterTabs } from '@/components/notifications/notification-filter-tabs';
import { NotificationToolbar } from '@/components/notifications/notification-toolbar';
import { EmptyState } from '@/components/notifications/empty-state';
import { useWebSocket } from '@/hooks/use-websocket';
import api from '@/lib/api';

type NotificationType = 'all' | 'comment' | 'interaction' | 'system';

export default function NotificationsPage() {
  const [filter, setFilter] = useState<NotificationType>('all');
  const queryClient = useQueryClient();

  const { data: notifications } = useQuery({
    queryKey: ['notifications', filter],
    queryFn: () => api.get('/notifications', {
      params: { type: filter === 'all' ? undefined : filter }
    }).then(r => r.data),
  });

  const { data: unreadCount } = useQuery({
    queryKey: ['notifications-unread-count'],
    queryFn: () => api.get('/notifications/unread-count').then(r => r.data.count),
  });

  // WebSocket 实时接收新通知
  useWebSocket('notifications', (newNotification) => {
    queryClient.invalidateQueries({ queryKey: ['notifications'] });
    queryClient.invalidateQueries({ queryKey: ['notifications-unread-count'] });
  });

  const markAsRead = useMutation({
    mutationFn: (id: string) => api.put(`/notifications/${id}/read`),
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ['notifications'] });
      queryClient.invalidateQueries({ queryKey: ['notifications-unread-count'] });
    },
  });

  const markAllAsRead = useMutation({
    mutationFn: () => api.put('/notifications/read-all'),
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ['notifications'] });
      queryClient.invalidateQueries({ queryKey: ['notifications-unread-count'] });
    },
  });

  // ...渲染逻辑
}
```

### 2.3 数据结构

```typescript
interface Notification {
  id: string;
  type: 'comment' | 'interaction' | 'update' | 'system';
  title: string;
  content: string;
  isRead: boolean;
  data: {
    novelId?: string;
    chapterId?: string;
    commentId?: string;
    userId?: string;
    userName?: string;
  };
  createdAt: string;
}

interface NotificationSettings {
  commentReply: boolean;
  workInteraction: boolean;
  updateReminder: boolean;
  systemAnnouncement: boolean;
  doNotDisturb: {
    enabled: boolean;
    startTime: string; // "22:00"
    endTime: string;   // "08:00"
  };
}
```

## 3. API 接口设计

### 3.1 获取通知列表
```http
GET /api/v1/notifications?type={type}&page={page}
```

**响应格式**:
```json
{
  "items": [
    {
      "id": "uuid",
      "type": "comment",
      "title": "评论回复",
      "content": "用户A 回复了你的评论",
      "isRead": false,
      "data": {
        "novelId": "uuid",
        "commentId": "uuid",
        "userId": "uuid",
        "userName": "用户A"
      },
      "createdAt": "2024-01-01T10:00:00Z"
    }
  ],
  "total": 50,
  "unreadCount": 5
}
```

### 3.2 获取未读数量
```http
GET /api/v1/notifications/unread-count
```

### 3.3 标记已读
```http
PUT /api/v1/notifications/{notificationId}/read
```

### 3.4 标记全部已读
```http
PUT /api/v1/notifications/read-all
```

### 3.5 获取通知设置
```http
GET /api/v1/notifications/settings
```

### 3.6 更新通知设置
```http
PUT /api/v1/notifications/settings
{
  "commentReply": true,
  "workInteraction": true,
  "updateReminder": true,
  "systemAnnouncement": true,
  "doNotDisturb": {
    "enabled": true,
    "startTime": "22:00",
    "endTime": "08:00"
  }
}
```
