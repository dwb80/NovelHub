## 2. 详细设计

### 2.1 页面组件设计

**文件**: `apps/frontend/src/app/bookshelf/page.tsx`

```typescript
'use client';

import { useState } from 'react';
import { useQuery } from '@tanstack/react-query';
import { Header } from '@/components/layout/header';
import { Footer } from '@/components/layout/footer';
import { BookshelfGrid } from '@/components/bookshelf/bookshelf-grid';
import { BookshelfList } from '@/components/bookshelf/bookshelf-list';
import { ViewToggle } from '@/components/bookshelf/view-toggle';
import { SortSelect } from '@/components/bookshelf/sort-select';
import { FilterTabs } from '@/components/bookshelf/filter-tabs';
import { useAuth } from '@/hooks/use-auth';
import api from '@/lib/api';

export default function BookshelfPage() {
  const { isAuthenticated } = useAuth();
  const [viewMode, setViewMode] = useState<'grid' | 'list'>('grid');
  const [sortBy, setSortBy] = useState<'recent' | 'added' | 'title'>('recent');
  const [filter, setFilter] = useState<'all' | 'reading' | 'completed' | 'unread'>('all');

  const { data, isLoading } = useQuery({
    queryKey: ['bookshelf', sortBy, filter],
    queryFn: async () => {
      const response = await api.get('/bookshelf', {
        params: { sort: sortBy, filter }
      });
      return response.data;
    },
    enabled: isAuthenticated,
  });

  // ...渲染逻辑
}
```

## 3. API 接口设计

### 3.1 获取书架列表
```http
GET /api/v1/bookshelf?sort={sort}&filter={filter}
```

**响应格式**:
```json
{
  "items": [
    {
      "id": "uuid",
      "novel": {
        "id": "uuid",
        "title": "小说标题",
        "authorName": "AI智能体作家名",
        "coverImage": "https://...",
        "totalChapters": 100,
        "status": "ONGOING"
      },
      "lastReadChapter": {
        "id": "uuid",
        "title": "第10章 xxx",
        "number": 10
      },
      "progress": 60.5,
      "isFinished": false,
      "lastReadAt": "2026-04-17T10:00:00Z",
      "addedAt": "2026-04-01T08:00:00Z",
      "hasUpdate": true,
      "unreadCount": 5
    }
  ],
  "total": 10
}
```

### 3.2 添加收藏
```http
POST /api/v1/bookshelf
{
  "novelId": "uuid"
}
```

### 3.3 取消收藏
```http
DELETE /api/v1/bookshelf/{novelId}
```

### 3.4 更新阅读进度
```http
PUT /api/v1/bookshelf/progress
{
  "novelId": "uuid",
  "chapterId": "uuid",
  "progress": 65.5
}
```

## 4. 状态管理

### 4.1 本地状态
- `viewMode`: 视图模式（grid/list）
- `sortBy`: 排序方式
- `filter`: 筛选条件

### 4.2 服务端状态
- 使用 React Query 管理书架数据
- 缓存时间：5分钟
- 自动刷新：窗口聚焦时

---

## 5. 性能优化

### 5.1 虚拟列表
- 书架超过20本时启用虚拟滚动
- 使用 react-window 或 react-virtualized

### 5.2 图片优化
- 封面图使用 Next.js Image 组件
- 懒加载：视口外图片延迟加载
- 占位符：加载时显示骨架屏

### 5.3 数据预取
- 鼠标悬停小说卡片时预取详情数据
- 路由预取：Link 组件自动预取

## 6. 错误处理

| 错误场景 | 处理方式 |
|---------|---------|
| 未登录 | 重定向到登录页 |
| 加载失败 | 显示重试按钮 |
| 空书架 | 显示引导提示 |
| 网络错误 | Toast提示 |
