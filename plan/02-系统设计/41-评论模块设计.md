## 2. 详细设计

### 2.1 评论区域组件

**文件**: `apps/frontend/src/components/comments/comment-section.tsx`

```typescript
'use client';

import { useState } from 'react';
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
import { CommentList } from './comment-list';
import { CommentInput } from './comment-input';
import { CommentSortTabs } from './comment-sort-tabs';
import { useAuth } from '@/hooks/use-auth';
import api from '@/lib/api';

interface CommentSectionProps {
  novelId: string;
  chapterId?: string; // 可选，章节评论
}

export function CommentSection({ novelId, chapterId }: CommentSectionProps) {
  const { isAuthenticated, user } = useAuth();
  const queryClient = useQueryClient();
  const [sortBy, setSortBy] = useState<'newest' | 'hottest'>('newest');

  const { data: comments, isLoading } = useQuery({
    queryKey: ['comments', novelId, chapterId, sortBy],
    queryFn: () => api.get('/comments', {
      params: { novelId, chapterId, sort: sortBy }
    }).then(r => r.data),
  });

  const postComment = useMutation({
    mutationFn: (content: string) => api.post('/comments', {
      novelId,
      chapterId,
      content,
    }),
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ['comments', novelId] });
    },
  });

  // ...渲染逻辑
}
```

### 2.3 数据结构

```typescript
interface Comment {
  id: string;
  novelId: string;
  chapterId?: string;
  author: {
    id: string;
    displayName: string;
    avatar: string;
  };
  content: string;
  likes: number;
  isLiked: boolean;
  replyCount: number;
  replies?: Reply[];
  createdAt: string;
  updatedAt: string;
}

interface Reply {
  id: string;
  commentId: string;
  author: {
    id: string;
    displayName: string;
    avatar: string;
  };
  replyTo?: {
    id: string;
    displayName: string;
  };
  content: string;
  likes: number;
  isLiked: boolean;
  createdAt: string;
}

interface CommentInputData {
  content: string;
  parentId?: string; // 回复的评论ID
  replyToUserId?: string; // 回复给哪个用户
}
```

## 3. API 接口设计

### 3.1 获取评论列表
```http
GET /api/v1/comments?novelId={novelId}&chapterId={chapterId}&sort={sort}&page={page}
```

**响应格式**:
```json
{
  "items": [
    {
      "id": "uuid",
      "author": {
        "id": "uuid",
        "displayName": "用户A",
        "avatar": "https://..."
      },
      "content": "评论内容",
      "likes": 12,
      "isLiked": false,
      "replyCount": 3,
      "createdAt": "2024-01-01T10:00:00Z"
    }
  ],
  "total": 128,
  "page": 1
}
```

### 3.2 发表评论
```http
POST /api/v1/comments
{
  "novelId": "uuid",
  "chapterId": "uuid",
  "content": "评论内容",
  "parentId": null
}
```

### 3.3 发表回复
```http
POST /api/v1/comments
{
  "novelId": "uuid",
  "chapterId": "uuid",
  "content": "回复内容",
  "parentId": "评论ID",
  "replyToUserId": "被回复用户ID"
}
```

### 3.4 点赞评论
```http
POST /api/v1/comments/{commentId}/like
```

### 3.5 取消点赞
```http
DELETE /api/v1/comments/{commentId}/like
```

### 3.6 删除评论
```http
DELETE /api/v1/comments/{commentId}
```

## 4. 交互设计

### 4.1 评论输入
- 最小长度：10字符
- 最大长度：500字符
- 支持表情选择
- 支持@用户（输入@弹出用户搜索）

### 4.2 楼中楼回复
- 最多显示3条最新回复
- 点击"查看更多回复"展开全部
- 支持回复特定用户

### 4.3 无限滚动
- 初始加载10条评论
- 滚动到底部自动加载更多
- 显示加载状态
