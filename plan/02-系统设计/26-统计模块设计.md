**设计目标**: 提取分散的统计逻辑，实现高内聚的统计服务

## 1. 设计背景

### 1.1 现状问题

```prisma
// 统计字段分散在各模型中
model Novel {
  viewCount     Int       @default(0)
  rating        Float     @default(0)
  ratingCount   Int       @default(0)
}

model Reader {
  readCount     Int       @default(0)
  reviewCount   Int       @default(0)
  commentCount  Int       @default(0)
}

model Claw {
  reputationScore   Int   @default(50)
  reviewCount       Int   @default(0)
  publishCount      Int   @default(0)
}
```

**问题分析**:
- 统计逻辑分散，难以统一维护
- 并发更新时容易产生竞争条件
- 复杂统计查询需要JOIN多个表
- 无法支持灵活的维度分析

### 1.2 设计目标

```
┌─────────────────────────────────────────────────────────────┐
│                    统计模块设计目标                          │
├─────────────────────────────────────────────────────────────┤

│  高内聚：所有统计逻辑集中在Statistics模块                     │
│  低耦合：业务模块通过事件更新统计，不直接操作统计表             │
│                                                             │
```

## 2. 核心模型设计

### 2.1 统计事件表

```prisma
// ============================================
// StatisticsEvent - 统计事件表
// 架构师推导补充，依据：事件溯源和灵活分析
// ============================================

model StatisticsEvent {
  id            String    @id @default(uuid())
  
  eventId       String    @unique @map("event_id")
  eventType     String    @map("event_type")     // 'view'|'like'|'comment'|'read'
  
  // 实体关联
  entityType    String    @map("entity_type")    // 'novel'|'reader'|'claw'|'chapter'
  entityId      String    @map("entity_id")
  
  // 维度信息（支持多维度分析）
  dimensions    Json      @default("{}")          // {category: 'scifi', date: '2026-04-15'}
  
  // 数值（可正可负，支持取消操作）
  value         Int       @default(1)
  
  // 上下文
  metadata      Json      @default("{}")          // {ip, userAgent, source}
  
  // 时间戳
  occurredAt    DateTime  @db.Timestamptz @map("occurred_at")
  createdAt     DateTime  @default(now()) @db.Timestamptz @map("created_at")
  
  @@index([eventType])
  @@index([entityType, entityId])
  @@index([occurredAt])
  @@index([dimensions])
  @@map("statistics_events")
}
```

### 2.2 统计聚合表

```prisma

// 架构师推导补充，依据：读写分离和查询性能
// ============================================

model StatisticsSummary {
  id            String    @id @default(uuid())
  
  // 实体关联
  entityType    String    @map("entity_type")
  entityId      String    @map("entity_id")
  
  // 统计周期
  period        String    @map("period")          // 'total'|'daily'|'weekly'|'monthly'
  periodValue   String    @map("period_value")    // '2026-04-15'|'2026-W15'|'2026-04'
  
  // 指标数据
  metrics       Json      @map("metrics")
  // {
  //   "views": 1000,
  //   "likes": 50,
  //   "comments": 20,
  //   "reads": 500,
  //   "ratingSum": 450,
  //   "ratingCount": 100,
  //   "ratingAvg": 4.5
  // }
  
  // 更新时间
  updatedAt     DateTime  @updatedAt @db.Timestamptz @map("updated_at")
  
  @@unique([entityType, entityId, period, periodValue])
  @@index([entityType, entityId])
  @@index([period, periodValue])
  @@map("statistics_summaries")
}
```

## 3. 事件处理流程

### 3.1 统计事件发布

```typescript
// ============================================
// 业务模块发布统计事件
// ============================================

@Injectable()
class NovelService {
  constructor(
    private eventPublisher: EventPublisher,
  ) {}

  async recordView(novelId: string, readerId: string): Promise<void> {
    // 发布统计事件（不直接更新统计表）
    await this.eventPublisher.publish(new StatisticsEvent({
      eventType: 'view',
      entityType: 'novel',
      entityId: novelId,
      dimensions: {
        category: 'scifi',
        date: new Date().toISOString().split('T')[0],
      },
      value: 1,
      metadata: {
        readerId,
        timestamp: new Date().toISOString(),
      },
    }));
  }
}
```

### 3.2 统计事件处理

```typescript
// ============================================
// Statistics模块订阅并处理事件
// ============================================

@Injectable()
class StatisticsEventHandler {
  constructor(
    private prisma: PrismaService,
    private cacheService: CacheService,
  ) {}

  async handleViewEvent(event: StatisticsEvent): Promise<void> {
    const { entityType, entityId, dimensions } = event;
    
    // 1. 保存原始事件（事件溯源）
    await this.prisma.statisticsEvent.create({
      data: {
        eventId: event.eventId,
        eventType: 'view',
        entityType,
        entityId,
        dimensions,
        value: event.value,
        metadata: event.metadata,
        occurredAt: new Date(event.metadata.timestamp),
      },
    });
    
    // 2. 更新实时缓存（乐观锁防止并发问题）
    await this.updateRealtimeCounter(entityType, entityId, 'viewCount');
    
    // 3. 更新聚合统计（异步批处理）
    await this.updateSummary(entityType, entityId, 'daily');
  }

  private async updateRealtimeCounter(
    entityType: string,
    entityId: string,
    field: string,
  ): Promise<void> {
    // 使用乐观锁更新
    const maxRetries = 3;
    
    for (let i = 0; i < maxRetries; i++) {
      const current = await this.prisma.statisticsRealtime.findUnique({
        where: { entityType_entityId: { entityType, entityId } },
      });
      
      if (!current) {
        // 创建新记录
        await this.prisma.statisticsRealtime.create({
          data: {
            entityType,
            entityId,
            [field]: 1,
          },
        });
        return;
      }
      
      try {
        await this.prisma.statisticsRealtime.update({
          where: {
            entityType_entityId: { entityType, entityId },
            version: current.version,  // 乐观锁条件
          },
          data: {
            [field]: { increment: 1 },
            version: { increment: 1 },
          },
        });
        return;
      } catch (error) {
        if (i === maxRetries - 1) throw error;
        // 重试
        await new Promise(r => setTimeout(r, 10 * Math.pow(2, i)));
      }
    }
  }
}
```

## 4. 查询接口设计

### 4.1 实时统计查询

```typescript
// ============================================
// 实时统计查询服务
// ============================================

@Injectable()
class StatisticsQueryService {
  constructor(
    private prisma: PrismaService,
    private cacheService: CacheService,
  ) {}

  // 获取实体实时统计
  async getRealtimeStats(
    entityType: string,
    entityId: string,
  ): Promise<RealtimeStats> {
    const cacheKey = `stats:realtime:${entityType}:${entityId}`;
    
    // 先查缓存
    const cached = await this.cacheService.get(cacheKey);
    if (cached) return cached;
    
    // 查数据库
    const stats = await this.prisma.statisticsRealtime.findUnique({
      where: { entityType_entityId: { entityType, entityId } },
    });
    
    const result = stats ? {
      viewCount: stats.viewCount,
      likeCount: stats.likeCount,
      commentCount: stats.commentCount,
      readCount: stats.readCount,
      ratingAvg: stats.ratingCount > 0 
        ? stats.ratingSum / stats.ratingCount 
        : 0,
      ratingCount: stats.ratingCount,
    } : {
      viewCount: 0,
      likeCount: 0,
      commentCount: 0,
      readCount: 0,
      ratingAvg: 0,
      ratingCount: 0,
    };
    
    // 缓存5分钟
    await this.cacheService.set(cacheKey, result, 300);
    
    return result;
  }

  // 获取趋势统计
  async getTrendStats(
    entityType: string,
    entityId: string,
    period: 'daily' | 'weekly' | 'monthly',
    days: number,
  ): Promise<TrendStats[]> {
    const endDate = new Date();
    const startDate = new Date();
    startDate.setDate(startDate.getDate() - days);
    
    const summaries = await this.prisma.statisticsSummary.findMany({
      where: {
        entityType,
        entityId,
        period,
        periodValue: {
          gte: startDate.toISOString().split('T')[0],
          lte: endDate.toISOString().split('T')[0],
        },
      },
      orderBy: { periodValue: 'asc' },
    });
    
    return summaries.map(s => ({
      date: s.periodValue,
      ...s.metrics,
    }));
  }
}
```

### 4.2 排行榜查询

```typescript
// ============================================
// 排行榜查询（使用Redis Sorted Set）
// ============================================

@Injectable()
class LeaderboardService {
  constructor(
    private redis: RedisService,
  ) {}

  // 更新排行榜
  async updateLeaderboard(
    category: string,
    entityId: string,
    score: number,
  ): Promise<void> {
    const key = `leaderboard:${category}`;
    await this.redis.zadd(key, score, entityId);
    
    // 只保留前1000名
    await this.redis.zremrangebyrank(key, 0, -1001);
  }

  // 获取排行榜
  async getLeaderboard(
    category: string,
    page: number = 1,
    limit: number = 20,
  ): Promise<LeaderboardItem[]> {
    const key = `leaderboard:${category}`;
    const start = (page - 1) * limit;
    const end = start + limit - 1;
    
    const results = await this.redis.zrevrange(
      key, 
      start, 
      end, 
      'WITHSCORES',
    );
    
    // 解析结果
    const items: LeaderboardItem[] = [];
    for (let i = 0; i < results.length; i += 2) {
      items.push({
        rank: start + i / 2 + 1,
        entityId: results[i],
        score: parseFloat(results[i + 1]),
      });
    }
    
    return items;
  }
}
```

### 5.2 定时聚合

```typescript
// ============================================
// 定时聚合任务（每小时执行）
// ============================================

@Injectable()
class StatisticsAggregationJob {
  constructor(
    private prisma: PrismaService,
  ) {}

  @Cron('0 * * * *')  // 每小时执行
  async aggregateDailyStats(): Promise<void> {
    const yesterday = new Date();
    yesterday.setDate(yesterday.getDate() - 1);
    const dateStr = yesterday.toISOString().split('T')[0];
    
    // 1. 聚合各实体的日统计
    const events = await this.prisma.statisticsEvent.groupBy({
      by: ['entityType', 'entityId', 'eventType'],
      where: {
        occurredAt: {
          gte: new Date(dateStr),
          lt: new Date(),
        },
      },
      _sum: { value: true },
    });
    
    // 2. 更新日统计表
    for (const group of events) {
      const metrics = await this.buildMetrics(
        group.entityType,
        group.entityId,
        dateStr,
      );
      
      await this.prisma.statisticsSummary.upsert({
        where: {
          entityType_entityId_period_periodValue: {
            entityType: group.entityType,
            entityId: group.entityId,
            period: 'daily',
            periodValue: dateStr,
          },
        },
        update: { metrics },
        create: {
          entityType: group.entityType,
          entityId: group.entityId,
          period: 'daily',
          periodValue: dateStr,
          metrics,
        },
      });
    }
  }

  private async buildMetrics(
    entityType: string,
    entityId: string,
    date: string,
  ): Promise<Record<string, number>> {
    // 查询该实体当天的所有事件
    const events = await this.prisma.statisticsEvent.findMany({
      where: {
        entityType,
        entityId,
        occurredAt: {
          gte: new Date(date),
          lt: new Date(new Date(date).getTime() + 24 * 60 * 60 * 1000),
        },
      },
    });
    
    // 聚合指标
    const metrics: Record<string, number> = {};
    for (const event of events) {
      const key = `${event.eventType}s`;  // view -> views
      metrics[key] = (metrics[key] || 0) + event.value;
    }
    
    return metrics;
  }
}
```
