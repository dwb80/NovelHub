## 1. 设计原则

### 1.1 高内聚低耦合原则

```
┌─────────────────────────────────────────────────────────────┐
│                     领域事件设计目标                          │
├─────────────────────────────────────────────────────────────┤
│                                                             │
│  高内聚：每个事件只包含该领域状态变更的必要信息                 │
│  低耦合：模块间不直接调用，通过事件总线通信                     │
│                                                             │
│  模块A ──发布事件──┐                                        │
│                    ▼                                        │
│              ┌──────────┐                                   │
│              │ 事件总线  │  ← 解耦点                          │
│              └──────────┘                                   │
│                    ▲                                        │
│  模块B ──订阅事件──┘                                        │
│                                                             │
└─────────────────────────────────────────────────────────────┘
```

### 1.2 事件分类

| 类别 | 说明 | 示例 | 持久化 |
|-----|------|------|--------|
| **领域事件** | 业务状态变更 | NovelCreated, ChapterPublished | ✅ 必须 |
| **集成事件** | 跨模块通信 | ReviewTaskCreated | ✅ 必须 |
| **通知事件** | 用户通知 | CommentReplied | ⚠️ 可选 |

### 2.1 基础事件接口

```typescript
// ============================================
// 领域事件基础接口
// 架构师推导补充，依据：事件溯源和幂等性要求
// ============================================

interface DomainEvent {
  // 事件元数据
  eventId: string;           // 全局唯一事件ID（UUID v4）
  eventType: string;         // 事件类型（大驼峰命名）
  aggregateId: string;       // 聚合根ID
  aggregateType: string;     // 聚合类型
  version: string;           // 事件版本（语义化版本）
  
  // 时间戳
  occurredAt: string;        // ISO 8601格式
  publishedAt: string;       // 发布到总线的时间
  
  // 上下文
  payload: EventPayload;     // 事件载荷（具体数据）
  metadata: EventMetadata;   // 元数据（追踪信息）
}

interface EventMetadata {
  correlationId: string;     // 关联ID（用于追踪请求链）
  causationId?: string;      // 因果ID（导致此事件的事件ID）
  userId?: string;           // 触发事件的用户ID
  source: string;            // 事件来源（服务名）
  ip?: string;               // 客户端IP
  userAgent?: string;        // 客户端UA
}
```

## 3. 发布/订阅机制

### 3.1 事件发布规范

```typescript
// ============================================
// 事件发布服务
// 架构师推导补充，依据：至少一次投递保证
// ============================================

@Injectable()
class EventPublisher {
  constructor(
    private prisma: PrismaService,
    private bullMq: BullMqService,
    private logger: Logger,
  ) {}

  async publish<T extends DomainEvent>(
    event: T,
    options: PublishOptions = {},
  ): Promise<void> {
    const { delay = 0, priority = 5 } = options;
    
    // 1. 生成事件ID（如果不存在）
    const eventId = event.eventId || uuidv4();
    
    // 2. 持久化到数据库（保证不丢失）
    await this.prisma.domainEvent.create({
      data: {
        eventId,
        eventType: event.eventType,
        aggregateId: event.aggregateId,
        aggregateType: event.aggregateType,
        payload: event.payload,
        metadata: event.metadata,
        status: 'PENDING',
        retryCount: 0,
        publishedAt: new Date(),
      },
    });
    
    // 3. 发送到消息队列
    await this.bullMq.publish('domain-events', event, {
      delay,
      priority,
      jobId: eventId,  // 用于幂等性
    });
    
    this.logger.log(`Event published: ${event.eventType} (${eventId})`);
  }
}
```

### 3.2 事件订阅规范

```typescript
// ============================================
// 事件处理器基类
// 架构师推导补充，依据：幂等性保证和错误处理
// ============================================

abstract class EventHandler<T extends DomainEvent> {
  abstract readonly eventType: string;
  
  constructor(
    private prisma: PrismaService,
    private logger: Logger,
  ) {}

  async handle(event: T): Promise<void> {
    // 1. 检查是否已处理（幂等性）
    const existing = await this.prisma.domainEvent.findUnique({
      where: { eventId: event.eventId },
    });
    
    if (existing?.status === 'COMPLETED') {
      this.logger.warn(`Event already processed: ${event.eventId}`);
      return;
    }
    
    // 2. 更新状态为处理中
    await this.prisma.domainEvent.update({
      where: { eventId: event.eventId },
      data: { status: 'PROCESSING' },
    });
    
    try {
      // 3. 执行业务逻辑
      await this.process(event);
      
      // 4. 标记为完成
      await this.prisma.domainEvent.update({
        where: { eventId: event.eventId },
        data: { 
          status: 'COMPLETED',
          processedAt: new Date(),
        },
      });
      
      this.logger.log(`Event processed: ${event.eventType} (${event.eventId})`);
    } catch (error) {
      // 5. 错误处理
      await this.handleError(event, error);
    }
  }
  
  abstract process(event: T): Promise<void>;
  
  private async handleError(event: T, error: Error): Promise<void> {
    const eventRecord = await this.prisma.domainEvent.findUnique({
      where: { eventId: event.eventId },
    });
    
    const retryCount = eventRecord?.retryCount || 0;
    const maxRetries = 3;
    
    if (retryCount < maxRetries) {
      // 重试
      await this.prisma.domainEvent.update({
        where: { eventId: event.eventId },
        data: {
          status: 'PENDING',
          retryCount: retryCount + 1,
          errorMessage: error.message,
        },
      });
      
      // 延迟重试（指数退避）
      const delay = Math.pow(2, retryCount) * 1000;
      await this.bullMq.publish('domain-events', event, { delay });
    } else {
      // 进入死信队列
      await this.prisma.domainEvent.update({
        where: { eventId: event.eventId },
        data: {
          status: 'DEAD_LETTER',
          errorMessage: error.message,
        },
      });
      
      this.logger.error(`Event failed after ${maxRetries} retries: ${event.eventId}`);
    }
  }
}

// ============================================
// 具体处理器示例：章节发布处理器
// ============================================

@Injectable()
class ChapterPublishedHandler extends EventHandler<ChapterPublishedEvent> {
  readonly eventType = 'ChapterPublished';
  
  constructor(
    prisma: PrismaService,
    logger: Logger,
    private reviewService: ReviewService,
    private nefService: NEFService,
  ) {
    super(prisma, logger);
  }
  
  async process(event: ChapterPublishedEvent): Promise<void> {
    // 1. 创建评审任务
    await this.reviewService.createTask({
      novelId: event.payload.novelId,
      chapterId: event.payload.chapterId,
      type: 'CHAPTER',
    });
    
    // 2. 触发NEF进化分析
    await this.nefService.analyzeChapter({
      chapterId: event.payload.chapterId,
      authorId: event.payload.authorId,
    });
  }
}
```

### 4.1 事件存储表结构

```prisma
// ============================================
// 领域事件存储表
// 架构师推导补充，依据：事件溯源和审计需求
// ============================================

model DomainEvent {
  id            String    @id @default(uuid())
  
  eventId       String    @unique @map("event_id")
  eventType     String    @map("event_type")
  aggregateId   String    @map("aggregate_id")
  aggregateType String    @map("aggregate_type")
  version       String    @default("1.0.0")
  
  payload       Json
  metadata      Json      @default("{}")
  
  status        EventStatus @default(PENDING)
  retryCount    Int       @default(0) @map("retry_count")
  errorMessage  String?   @map("error_message")
  
  publishedAt   DateTime  @default(now()) @db.Timestamptz @map("published_at")
  processedAt   DateTime? @db.Timestamptz @map("processed_at")
  
  @@index([eventType])
  @@index([aggregateId])
  @@index([status])
  @@index([publishedAt])
  @@map("domain_events")
}

enum EventStatus {
  PENDING     // 待处理
  PROCESSING  // 处理中
  COMPLETED   // 已完成
  FAILED      // 失败（可重试）
  DEAD_LETTER // 死信（多次失败后）
}
```

## 5. 失败重试机制

### 5.1 重试策略

```typescript
// ============================================
// 指数退避重试策略
// 架构师推导补充，依据：避免雪崩效应
// ============================================

interface RetryPolicy {
  maxRetries: number;        // 最大重试次数（默认3次）
  baseDelay: number;         // 基础延迟（毫秒，默认1000）
  maxDelay: number;          // 最大延迟（毫秒，默认30000）
  backoffMultiplier: number; // 退避倍数（默认2）
}

function calculateDelay(
  retryCount: number,
  policy: RetryPolicy,
): number {
  const exponentialDelay = policy.baseDelay * Math.pow(
    policy.backoffMultiplier,
    retryCount,
  );
  
  // 添加抖动（避免同时重试）
  const jitter = Math.random() * 1000;
  
  return Math.min(exponentialDelay + jitter, policy.maxDelay);
}

// 重试延迟示例：
// 第1次重试: 1000ms + jitter
// 第2次重试: 2000ms + jitter
// 第3次重试: 4000ms + jitter
```
