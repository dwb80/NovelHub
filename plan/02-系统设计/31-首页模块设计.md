## 3. 数据获取设计

### 3.1 小说列表数据获取

**文件**: `apps/frontend/src/components/novel/novel-grid.tsx`

```typescript
'use client';

import { useQuery } from '@tanstack/react-query';
import { NovelCard } from './novel-card';
import { api } from '@/lib/api';

interface Novel {
  id: string;
  title: string;
  cover: string;
  authorName: string;
  tags: string[];
  viewCount: number;
  likeCount: number;
  wordCount: number;
}

export function NovelGrid() {
  const { data, isLoading } = useQuery({
    queryKey: ['novels', 'popular'],
    queryFn: async () => {
      const response = await api.get('/novels', {
        params: { page: 1, limit: 12, sort: 'viewCount' },
      });
      return response.data;
    },
  });

  if (isLoading) {
    return <NovelGridSkeleton />;
  }

  return (
    <div className="grid grid-cols-2 md:grid-cols-4 lg:grid-cols-6 gap-4">
      {data?.novels?.map((novel: Novel) => (
        <NovelCard key={novel.id} novel={novel} />
      ))}
    </div>
  );
}
```

## 4. 性能优化

### 4.1 加载优化

| 优化策略 | 实现方式 | 效果 |
|---------|---------|------|
| 骨架屏 | NovelGridSkeleton组件 | 减少感知加载时间 |
| 图片懒加载 | next/image 自动处理 | 减少首屏请求 |
| 数据预取 | React Query prefetch | 提升交互响应 |
| 组件分割 | Suspense + 动态导入 | 减少初始包体积 |

### 4.2 缓存策略

```typescript
// React Query 缓存配置
const queryClient = new QueryClient({
  defaultOptions: {
    queries: {
      staleTime: 5 * 60 * 1000, // 5分钟
      cacheTime: 10 * 60 * 1000, // 10分钟
      refetchOnWindowFocus: false,
    },
  },
});
```
