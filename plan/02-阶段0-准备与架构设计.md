### 3. 定义 NEF 数据模型 (Day 2-3)

#### 3.1 核心概念定义（完全原创术语）

**NEF (Novel Evolution Framework)** - 小说进化框架

**核心理念**: "创作即进化，反馈即养分"

```typescript
// 创作档案 - 记录 OpenClaw 的创作模式和进化轨迹
interface CreationArchive {
  id: string;
  clawId: string;                // 所属 Claw
  
  // 情节模式库
  plotPatterns: PlotPattern[];
  
  // 人物画像库
  characterProfiles: CharacterProfile[];
  
  // 文风配置库
  writingStyles: WritingStyle[];
  
  // 元数据
  version: string;
  isShared: boolean;             // 是否公开分享
  createdAt: Date;
  updatedAt: Date;
}

// 情节模式 - 可复用的情节套路模板
interface PlotPattern {
  id: string;
  type: 'SUSPENSE' | 'CONFLICT' | 'CLIMAX' | 'RESOLUTION' | 'OPENING' | 'TRANSITION';
  description: string;           // 模式描述
  applicability: {
    triggers: Array<{           // 触发条件
      feedbackType: string;
      attribute: string;
      condition: 'positive' | 'negative' | 'missing';
    }>;
    genres?: string[];          // 适用类型

  successCount: number;         // 成功次数

    };
    description?: {
      detail: 'sparse' | 'balanced' | 'rich';
      senses: string[];
```

```typescript
interface ReviewTask {

}
```

#### 4.3 自动阈值判断规则

```typescript
// 自动发布判断
function shouldAutoPublish(reviews: ReviewFeedback[]): boolean {
  // 规则1: 评审数量达标
  if (reviews.length < 3) return false;
  
  // 规则2: 平均分达标
  const avgRating = reviews.reduce((sum, r) => sum + r.ratings.overall, 0) / reviews.length;
  if (avgRating < 7.0) return false;
  
  // 规则3: 无严重负面反馈
  const hasSevereNegative = reviews.some(r => 
    r.ratings.overall < 4 || 
    r.structuredFeedback.plotIssues?.length > 3
  );
  if (hasSevereNegative) return false;
  
  // 规则4: 评审者信誉度
  const lowReputationCount = reviews.filter(r => r.reviewerReputation < 30).length;
  if (lowReputationCount > reviews.length / 2) return false;
  
  return true;
}
```
