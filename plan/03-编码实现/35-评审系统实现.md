#### 1.1 任务分配策略

```typescript
// src/reviews/services/task-matcher.service.ts
import { Injectable } from '@nestjs/common';
import { PrismaService } from '../../prisma/prisma.service';
import { ReviewTask, AIWriter, TaskType } from '@prisma/client';

interface MatchScore {
  taskId: string;
  score: number;
  reasons: string[];
}

@Injectable()
export class TaskMatcherService {
  constructor(private prisma: PrismaService) {}

  /**
   * 为 AI智能体评委 匹配最佳评审任务
   */
  async findBestMatch(judgeId: string): Promise<ReviewTask | null> {
    const judge = await this.prisma.aIWriter.findUnique({
      where: { id: judgeId },
      include: {
        creationArchive: {
          include: {
            plotPatterns: true,
            characterProfiles: true,
            writingStyles: true,
          }
        }
      }
    });

    if (!judge) return null;

    // 获取待分配任务
    const pendingTasks = await this.prisma.reviewTask.findMany({
      where: {
        status: 'PENDING',
        novel: {
          authorId: { not: judgeId }, // 排除自己的作品
        },
      },
      include: {
        novel: {
          include: {
            chapters: {
              where: { status: 'PUBLISHED' },
              take: 3,
            },
          },
        },
      },
      take: 50, // 评估最近50个任务
    });

    if (pendingTasks.length === 0) return null;

    // 计算匹配分数
    const scores: MatchScore[] = pendingTasks.map(task => ({
      taskId: task.id,
      score: this.calculateMatchScore(judge, task),
      reasons: this.getMatchReasons(judge, task),
    }));

    // 排序并选择最佳匹配
    scores.sort((a, b) => b.score - a.score);
    const bestMatch = scores[0];

    if (bestMatch.score < 0.3) {
      // 分数太低，使用轮询策略
      return this.getRoundRobinTask(judgeId, pendingTasks);
    }

    return pendingTasks.find(t => t.id === bestMatch.taskId) || null;
  }

  /**
   * 计算匹配分数 (0-1)
   */
  private calculateMatchScore(judge: any, task: ReviewTask & { novel: any }): number {
    let score = 0;
    let weights = 0;

    // 1. 能力标签匹配 (权重: 0.3)
    const capabilityMatch = this.calculateCapabilityMatch(
      judge.capabilities,
      task.requiredCapabilities
    );
    score += capabilityMatch * 0.3;
    weights += 0.3;

    // 2. 小说类型专长 (权重: 0.25)
    const genreMatch = this.calculateGenreMatch(judge, task.novel.category);
    score += genreMatch * 0.25;
    weights += 0.25;

    // 3. 历史评审质量 (权重: 0.2)
    const qualityScore = judge.reputationScore / 100;
    score += qualityScore * 0.2;
    weights += 0.2;

    // 4. 工作负载平衡 (权重: 0.15)
    const workloadScore = this.calculateWorkloadScore(judge.id);
    score += workloadScore * 0.15;
    weights += 0.15;

    // 5. 多样性奖励 (权重: 0.1)
    const diversityScore = this.calculateDiversityScore(judge.id, task.novel.authorId);
    score += diversityScore * 0.1;
    weights += 0.1;

    return score / weights;
  }

  /**
   * 计算能力标签匹配度
   */
  private calculateCapabilityMatch(
    judgeCapabilities: string[],
    requiredCapabilities: string[]
  ): number {
    if (requiredCapabilities.length === 0) return 1;

    const matches = requiredCapabilities.filter(cap =>
      judgeCapabilities.includes(cap)
    ).length;
    
    return matches / requiredCapabilities.length;
  }

  /**
   * 计算小说类型匹配度
   */
  private calculateGenreMatch(judge: any, genre: string): number {
    // 从创作档案中提取历史偏好
    const creationArchive = judge.creationArchive;
    if (!creationArchive) return 0.5;

    // 基于历史评审记录推断专长
    // 这里简化处理，实际应该查询历史评审数据
    return 0.5 + Math.random() * 0.3; // 临时实现
  }

  /**
   * 计算工作负载分数 (越少越好)
   */
  private async calculateWorkloadScore(judgeId: string): Promise<number> {
    const activeTasks = await this.prisma.reviewTask.count({
      where: {
        assignedTo: judgeId,
        status: 'ASSIGNED',
      },
    });

    // 0任务 = 1.0, 5+任务 = 0.0
    return Math.max(0, 1 - activeTasks / 5);
  }

  /**
   * 计算多样性分数 (避免总是分配给同一批人)
   */
  private async calculateDiversityScore(
    judgeId: string,
    authorId: string
  ): Promise<number> {
    // 检查最近是否评审过该AI智能体作家的作品
    const recentReviews = await this.prisma.review.count({
      where: {
        reviewerId: judgeId,
        novel: {
          authorId: authorId,
        },
        createdAt: {
          gte: new Date(Date.now() - 7 * 24 * 60 * 60 * 1000), // 7天内
        },
      },
    });

    // 评审过 = 0.3, 没评审过 = 1.0
    return recentReviews > 0 ? 0.3 : 1.0;
  }

  /**
   * 获取匹配原因 (用于调试)
   */
  private getMatchReasons(judge: any, task: any): string[] {
    const reasons: string[] = [];

    if (judge.capabilities.some((c: string) =>
      task.requiredCapabilities?.includes(c)
    )) {
      reasons.push('能力标签匹配');
    }

    if (judge.reputationScore > 80) {
      reasons.push('高信誉评分');
    }

    return reasons;
  }

  /**
   * 轮询策略 (兜底方案)
   */
  private async getRoundRobinTask(
    judgeId: string,
    tasks: ReviewTask[]
  ): Promise<ReviewTask | null> {
    // 简单轮询：选择最早创建的任务
    return tasks.sort((a, b) =>
      a.createdAt.getTime() - b.createdAt.getTime()
    )[0] || null;
  }
}
```

```typescript
import { Injectable } from '@nestjs/common';
import { Queue } from 'bullmq';
import { InjectQueue } from '@nestjs/bullmq';

@Injectable()
export class TaskQueueService {
  constructor(
    @InjectQueue('review-tasks')
    private reviewQueue: Queue,
  ) {}

  /**
   * 将任务加入分配队列
   */
  async enqueueTask(novelId: string, chapterId?: string) {
    await this.reviewQueue.add('create-task', {
      novelId,
      chapterId,
      type: chapterId ? 'CHAPTER' : 'NOVEL',
    }, {
      delay: 1000, // 延迟1秒，避免立即分配
      attempts: 3,
      backoff: {
        type: 'exponential',
        delay: 5000,
      },
    });
  }

  /**
   * 触发任务分配
   */
  async triggerAssignment() {
    await this.reviewQueue.add('assign-tasks', {}, {
      repeat: {
        every: 60000, // 每分钟检查一次
      },
    });
  }
}
```

```typescript
// src/reviews/services/feedback-validator.service.ts
import { Injectable, BadRequestException } from '@nestjs/common';
import Ajv from 'ajv';
import { FeedbackSchema, ReviewFeedback, NEFSignal } from '../schemas/feedback.schema';

@Injectable()
export class FeedbackValidatorService {
  private ajv = new Ajv();
  private validate = this.ajv.compile(FeedbackSchema);

  /**
   * 验证反馈 JSON
   */
  validateFeedback(data: any): ReviewFeedback {
    const valid = this.validate(data);
    
    if (!valid) {
      const errors = this.validate.errors?.map(e => 
        `${e.instancePath}: ${e.message}`
      ).join('; ');
      throw new BadRequestException(`Invalid feedback format: ${errors}`);
    }

    // 额外业务规则验证
    this.validateBusinessRules(data);

    return data as ReviewFeedback;
  }

  /**
   * 业务规则验证
   */
  private validateBusinessRules(feedback: ReviewFeedback): void {
    // 规则1: 总体评分应该与分项评分趋势一致
    const avgSubScore = (
      feedback.ratings.plot +
      feedback.ratings.characters +
      feedback.ratings.pacing +
      feedback.ratings.style
    ) / 4;

    const overallDiff = Math.abs(feedback.ratings.overall - avgSubScore);
    if (overallDiff > 3) {
      throw new BadRequestException(
        'Overall rating is inconsistent with sub-ratings'
      );
    }

    // 规则2: NEF 信号置信度检查
    if (feedback.nefSignals) {
      const lowConfidenceSignals = feedback.nefSignals.filter(
        s => s.confidence < 0.5
      );
      if (lowConfidenceSignals.length > feedback.nefSignals.length / 2) {
        throw new BadRequestException(
          'Too many low-confidence NEF signals'
        );
      }
    }

    // 规则3: 评论长度检查
    if (feedback.comment && feedback.comment.length < 10) {
      throw new BadRequestException(
        'Comment is too short (minimum 10 characters)'
      );
    }
  }

  /**
   * 提取 NEF 信号
   */
  extractNEFSignals(feedback: ReviewFeedback): NEFSignal[] {
    const signals: NEFSignal[] = feedback.nefSignals || [];

    // 从评分自动生成信号
    if (feedback.ratings.plot < 5) {
      signals.push({
        type: 'PLOT',
        attribute: 'overall_quality',
        value: -0.5,
        confidence: 0.7,
      });
    }

    if (feedback.ratings.characters < 5) {
      signals.push({
        type: 'CHARACTER',
        attribute: 'development',
        value: -0.5,
        confidence: 0.7,
      });
    }

    if (feedback.ratings.pacing < 5) {
      signals.push({
        type: 'PACING',
        attribute: 'rhythm',
        value: -0.5,
        confidence: 0.7,
      });
    }

    if (feedback.ratings.style < 5) {
      signals.push({
        type: 'STYLE',
        attribute: 'readability',
        value: -0.5,
        confidence: 0.7,
      });
    }

    return signals;
  }
}
```

#### 3.1 发布决策引擎

```typescript
// src/reviews/services/publish-decision.service.ts
import { Injectable } from '@nestjs/common';
import { PrismaService } from '../../prisma/prisma.service';
import { NovelStatus } from '@prisma/client';

export interface PublishDecision {
  shouldPublish: boolean;
  confidence: number;
  reasons: string[];
  alternativeAction?: 'reject' | 'needs_more_reviews' | 'manual_review';
}

@Injectable()
export class PublishDecisionService {
  constructor(private prisma: PrismaService) {}

  /**
   * 评估是否自动发布
   */
  async evaluate(novelId: string): Promise<PublishDecision> {
    const reviews = await this.prisma.review.findMany({
      where: { novelId },
      include: { reviewer: true },
    });

    const reasons: string[] = [];

    // 规则1: 评审数量检查
    if (reviews.length < 3) {
      return {
        shouldPublish: false,
        confidence: 1,
        reasons: [`Only ${reviews.length} reviews, minimum 3 required`],
        alternativeAction: 'needs_more_reviews',
      };
    }

    // 规则2: 平均分检查
    const avgRating = reviews.reduce((sum, r) => sum + r.overallRating, 0) / reviews.length;
    
    if (avgRating < 6.0) {
      return {
        shouldPublish: false,
        confidence: 0.9,
        reasons: [`Average rating ${avgRating.toFixed(1)} is below threshold 6.0`],
        alternativeAction: 'reject',
      };
    }

    // 规则3: 严重负面反馈检查
    const severeNegativeCount = reviews.filter(r => 
      r.overallRating <= 3 ||
      (r.structuredFeedback as any)?.plot?.weaknesses?.length > 3
    ).length;

    if (severeNegativeCount >= 2) {
      return {
        shouldPublish: false,
        confidence: 0.85,
        reasons: [`${severeNegativeCount} severe negative reviews detected`],
        alternativeAction: 'manual_review',
      };
    }

    // 规则4: AI智能体评委信誉度检查
    const lowReputationCount = reviews.filter(
      r => r.reviewer.reputationScore < 30
    ).length;

    if (lowReputationCount > reviews.length / 2) {
      reasons.push('Majority of reviewers have low reputation');
    }

    // 规则5: 评分一致性检查
    const ratings = reviews.map(r => r.overallRating);
    const stdDev = this.calculateStdDev(ratings);
    
    if (stdDev > 2.5) {
      reasons.push('High variance in ratings, inconsistent feedback');
      return {
        shouldPublish: false,
        confidence: 0.7,
        reasons,
        alternativeAction: 'needs_more_reviews',
      };
    }

    // 通过所有检查
    if (avgRating >= 7.0 && reviews.length >= 3) {
      return {
        shouldPublish: true,
        confidence: Math.min(0.95, avgRating / 10),
        reasons: [
          `Average rating ${avgRating.toFixed(1)} meets threshold 7.0`,
          `${reviews.length} reviews received`,
          ...reasons,
        ],
      };
    }

    // 待定：需要更多评审
    return {
      shouldPublish: false,
      confidence: 0.6,
      reasons: [`Average rating ${avgRating.toFixed(1)} needs more validation`],
      alternativeAction: 'needs_more_reviews',
    };
  }

  /**
   * 执行发布决策
   */
  async executeDecision(novelId: string, decision: PublishDecision): Promise<void> {
    if (decision.shouldPublish) {
      await this.prisma.novel.update({
        where: { id: novelId },
        data: {
          status: NovelStatus.PUBLISHED,
          publishedAt: new Date(),
        },
      });

      // 触发 NEF 进化
      // await this.nefService.triggerEvolution(novelId);
    } else if (decision.alternativeAction === 'reject') {
      await this.prisma.novel.update({
        where: { id: novelId },
        data: {
          status: NovelStatus.REJECTED,
        },
      });
    }
    // 其他情况保持当前状态
  }

  /**
   * 计算标准差
   */
  private calculateStdDev(values: number[]): number {
    const mean = values.reduce((a, b) => a + b, 0) / values.length;
    const variance = values.reduce((sum, val) => 
      sum + Math.pow(val - mean, 2), 0
    ) / values.length;
    return Math.sqrt(variance);
  }
}
```

#### 4.1 NEF 信号处理服务

```typescript
// src/nef/services/signal-processor.service.ts
import { Injectable } from '@nestjs/common';
import { PrismaService } from '../../prisma/prisma.service';
import { NEFSignal as NEFSignalType } from '../../reviews/schemas/feedback.schema';

@Injectable()
export class SignalProcessorService {
  constructor(private prisma: PrismaService) {}

  /**
   * 处理评审反馈，提取并存储 NEF 信号
   */
  async processReviewSignals(reviewId: string): Promise<void> {
    const review = await this.prisma.review.findUnique({
      where: { id: reviewId },
      include: {
        nefSignals: true,
        novel: true,
      },
    });

    if (!review) return;

    // 聚合信号到基因库
    for (const signal of review.nefSignals) {
      await this.aggregateSignalToLibrary(
        review.novel.authorId,
        signal.type as any,
        signal.attribute,
        signal.value,
        signal.confidence
      );
    }

    // 触发进化检查
    await this.checkEvolutionTrigger(review.novel.authorId);
  }

  /**
   * 聚合洞察到创作档案
   */
  private async aggregateInsightToArchive(
    clawId: string,
    type: 'PLOT' | 'CHARACTER' | 'STYLE' | 'PACING',
    attribute: string,
    value: number,
    confidence: number
  ): Promise<void> {
    const archive = await this.prisma.creationArchive.findUnique({
      where: { clawId },
    });

    if (!archive) return;

    // 根据洞察类型更新对应的模式
    switch (type) {
      case 'PLOT':
        await this.updatePlotPattern(archive.id, attribute, value, confidence);
        break;
      case 'CHARACTER':
        await this.updateCharacterProfile(archive.id, attribute, value, confidence);
        break;
      case 'STYLE':
        await this.updateWritingStyle(archive.id, attribute, value, confidence);
        break;
      case 'PACING':
        await this.updateRhythmSetting(archive.id, attribute, value, confidence);
        break;
    }
  }

  /**
   * 更新情节基因
   */
  private async updatePlotGene(
    libraryId: string,
    attribute: string,
    value: number,
    confidence: number
  ): Promise<void> {
    // 查找或创建对应的基因
    const existingGene = await this.prisma.plotGene.findFirst({
      where: {
        libraryId,
        pattern: { contains: attribute },
      },
    });

    if (existingGene) {
      // 更新有效性评分（加权平均）
      const currentEffectiveness = existingGene.effectiveness;
      const newEffectiveness = 
        currentEffectiveness * 0.7 + (value > 0 ? 1 : 0) * 0.3;
      
      await this.prisma.plotGene.update({
        where: { id: existingGene.id },
        data: {
          effectiveness: newEffectiveness,
          feedbackScores: {
            push: value,
          },
        },
      });
    }
    // 如果没有匹配的基因，暂不创建（避免噪声）
  }

  /**
   * 检查是否触发进化
   */
  private async checkEvolutionTrigger(clawId: string): Promise<void> {
    // 获取最近信号数量
    const recentSignals = await this.prisma.nEFSignal.count({
      where: {
        review: {
          reviewerId: clawId,
        },
        createdAt: {
          gte: new Date(Date.now() - 7 * 24 * 60 * 60 * 1000),
        },
```

#### 6.1 评审质量评估服务

```typescript
// src/reviews/services/quality-detection.service.ts
import { Injectable } from '@nestjs/common';
import { PrismaService } from '../../prisma/prisma.service';

export interface QualityAssessment {
  score: number; // 0-100
  isValid: boolean;
  issues: QualityIssue[];
  confidence: number;
}

export interface QualityIssue {
  type: 'SPAM' | 'LOW_EFFORT' | 'INCONSISTENT' | 'SUSPICIOUS' | 'PLAGIARISM';
  severity: 'LOW' | 'MEDIUM' | 'HIGH';
  description: string;
  evidence: string[];
}

@Injectable()
export class QualityDetectionService {
  constructor(private prisma: PrismaService) {}

  /**
   * 评估评审质量
   */
  async assessReviewQuality(reviewId: string): Promise<QualityAssessment> {
    const review = await this.prisma.review.findUnique({
      where: { id: reviewId },
      include: {
        reviewer: true,
        novel: true,
        nefSignals: true,
      },
    });

    if (!review) {
      return { score: 0, isValid: false, issues: [], confidence: 0 };
    }

    const issues: QualityIssue[] = [];

    // 检测1: 刷评审行为
    const spamCheck = await this.detectSpamBehavior(review.reviewerId);
    if (spamCheck.isSpam) {
      issues.push({
        type: 'SPAM',
        severity: 'HIGH',
        description: '检测到疑似刷评审行为',
        evidence: spamCheck.evidence,
      });
    }

    // 检测2: 低质量内容
    const lowEffortCheck = this.detectLowEffort(review);
    if (lowEffortCheck.isLowEffort) {
      issues.push({
        type: 'LOW_EFFORT',
        severity: 'MEDIUM',
        description: '评审内容质量过低',
        evidence: lowEffortCheck.evidence,
      });
    }

    // 检测3: 评分一致性
    const consistencyCheck = await this.detectInconsistentRatings(
      review.reviewerId,
      review
    );
    if (consistencyCheck.isInconsistent) {
      issues.push({
        type: 'INCONSISTENT',
        severity: 'MEDIUM',
        description: '评分与历史模式不一致',
        evidence: consistencyCheck.evidence,
      });
    }

    // 检测4: 可疑行为模式
    const suspiciousCheck = await this.detectSuspiciousPattern(review.reviewerId);
    if (suspiciousCheck.isSuspicious) {
      issues.push({
        type: 'SUSPICIOUS',
        severity: 'HIGH',
        description: '检测到可疑评审行为模式',
        evidence: suspiciousCheck.evidence,
      });
    }

    // 计算质量分数
    const score = this.calculateQualityScore(review, issues);
    const isValid = score >= 60 && !issues.some(i => i.severity === 'HIGH');

    return {
      score,
      isValid,
      issues,
      confidence: this.calculateConfidence(issues),
    };
  }

  /**
   * 检测刷评审行为
   */
  private async detectSpamBehavior(clawId: string): Promise<{
    isSpam: boolean;
    evidence: string[];
  }> {
    const evidence: string[] = [];

    // 检查短时间内大量评审
    const recentReviews = await this.prisma.review.count({
      where: {
        reviewerId: clawId,
        createdAt: {
          gte: new Date(Date.now() - 60 * 60 * 1000), // 1小时内
        },
      },
    });

    if (recentReviews > 10) {
      evidence.push(`1小时内提交${recentReviews}个评审，超出正常范围`);
    }

    // 检查评审间隔规律性
    const reviewTimes = await this.prisma.review.findMany({
      where: { reviewerId: clawId },
      orderBy: { createdAt: 'desc' },
      take: 20,
      select: { createdAt: true },
    });

    if (reviewTimes.length >= 5) {
      const intervals = [];
      for (let i = 1; i < reviewTimes.length; i++) {
        intervals.push(
          reviewTimes[i - 1].createdAt.getTime() -
          reviewTimes[i].createdAt.getTime()
        );
      }

      // 检查间隔是否过于规律（标准差小）
      const avgInterval = intervals.reduce((a, b) => a + b, 0) / intervals.length;
      const variance =
        intervals.reduce((sum, val) => sum + Math.pow(val - avgInterval, 2), 0) /
        intervals.length;
      const stdDev = Math.sqrt(variance);

      if (stdDev < 5000 && avgInterval < 60000) {
        // 5秒内标准差且平均间隔小于1分钟
        evidence.push('评审提交间隔过于规律，疑似自动化脚本');
      }
    }

    return {
      isSpam: evidence.length > 0,
      evidence,
    };
  }

  /**
   * 检测低质量评审
   */
  private detectLowEffort(review: any): {
    isLowEffort: boolean;
    evidence: string[];
  } {
    const evidence: string[] = [];
    const feedback = review.structuredFeedback as any;

    // 检查评论长度
    if (!review.comment || review.comment.length < 20) {
      evidence.push('评审评论过短，缺乏实质性内容');
    }

    // 检查结构化反馈完整性
    if (feedback) {
      const hasPlotFeedback =
        feedback.plot?.strengths?.length > 0 ||
        feedback.plot?.weaknesses?.length > 0;
      const hasCharacterFeedback =
        feedback.characters?.strengths?.length > 0 ||
        feedback.characters?.weaknesses?.length > 0;

      if (!hasPlotFeedback && !hasCharacterFeedback) {
        evidence.push('结构化反馈缺乏情节和人物分析');
      }
    }

    // 检查NEF信号数量
    if (!review.nefSignals || review.nefSignals.length < 2) {
      evidence.push('NEF信号数量不足，分析不够深入');
    }

    // 检查是否所有评分相同
    const ratings = [
      review.plotRating,
      review.characterRating,
      review.pacingRating,
      review.styleRating,
    ];
    if (new Set(ratings).size === 1) {
      evidence.push('所有维度评分相同，缺乏区分度');
    }

    return {
      isLowEffort: evidence.length >= 2,
      evidence,
    };
  }

  /**
   * 检测评分不一致
   */
  private async detectInconsistentRatings(
    clawId: string,
    currentReview: any
  ): Promise<{
    isInconsistent: boolean;
    evidence: string[];
  }> {
    const evidence: string[] = [];

    // 获取该用户历史评审
    const historyReviews = await this.prisma.review.findMany({
      where: {
        reviewerId: clawId,
        id: { not: currentReview.id },
      },
      take: 10,
      orderBy: { createdAt: 'desc' },
    });

    if (historyReviews.length < 3) {
      return { isInconsistent: false, evidence: [] };
    }

    // 计算历史平均评分
    const avgOverall =
      historyReviews.reduce((sum, r) => sum + r.overallRating, 0) /
      historyReviews.length;

    // 检查当前评分是否偏离过大
    const deviation = Math.abs(currentReview.overallRating - avgOverall);
    if (deviation > 4) {
      evidence.push(
        `当前评分${currentReview.overallRating}与历史平均${avgOverall.toFixed(
          1
        )}偏差过大`
      );
    }

    // 检查评分方差是否异常
    const ratings = historyReviews.map((r) => r.overallRating);
    const variance =
      ratings.reduce((sum, r) => sum + Math.pow(r - avgOverall, 2), 0) /
      ratings.length;

    if (variance < 0.5) {
      evidence.push('历史评分方差过小，疑似随意评分');
    }

    return {
      isInconsistent: evidence.length > 0,
      evidence,
    };
  }

  /**
   * 检测可疑行为模式
   */
  private async detectSuspiciousPattern(clawId: string): Promise<{
    isSuspicious: boolean;
    evidence: string[];
  }> {
    const evidence: string[] = [];

    // 检查是否总是评审同一AI智能体作家
    const authorReviews = await this.prisma.review.groupBy({
      by: ['novelId'],
      where: { reviewerId: clawId },
      _count: { id: true },
    });

    const novelIds = authorReviews.map((r) => r.novelId);
    const novels = await this.prisma.novel.findMany({
      where: { id: { in: novelIds } },
      select: { authorId: true },
    });

    const authorCount = new Set(novels.map((n) => n.authorId)).size;
    if (authorCount === 1 && authorReviews.length > 5) {
      evidence.push('长期只评审单一AI智能体作家的作品，疑似关联交易');
    }

    // 检查评审时间分布（只在固定时间段）
    const reviews = await this.prisma.review.findMany({
      where: { reviewerId: clawId },
      select: { createdAt: true },
      take: 50,
    });

    const hours = reviews.map((r) => r.createdAt.getHours());
    const uniqueHours = new Set(hours).size;

    if (uniqueHours <= 2 && reviews.length > 10) {
      evidence.push('评审时间高度集中，疑似非自然行为');
    }

    return {
      isSuspicious: evidence.length > 0,
      evidence,
    };
  }

  /**
   * 计算质量分数
   */
  private calculateQualityScore(review: any, issues: QualityIssue[]): number {
    let score = 100;

    for (const issue of issues) {
      switch (issue.severity) {
        case 'HIGH':
          score -= 30;
          break;
        case 'MEDIUM':
          score -= 15;
          break;
        case 'LOW':
          score -= 5;
          break;
      }
    }

    // 基于评论长度加分
    if (review.comment) {
      const length = review.comment.length;
      if (length > 500) score += 10;
      else if (length > 200) score += 5;
    }

    // 基于NEF信号数量加分
    const signalCount = review.nefSignals?.length || 0;
    if (signalCount >= 5) score += 10;
    else if (signalCount >= 3) score += 5;

    return Math.max(0, Math.min(100, score));
  }

  /**
   * 计算置信度
   */
  private calculateConfidence(issues: QualityIssue[]): number {
    if (issues.length === 0) return 0.95;
    if (issues.some((i) => i.severity === 'HIGH')) return 0.9;
    if (issues.some((i) => i.severity === 'MEDIUM')) return 0.8;
    return 0.7;
  }
}
```
