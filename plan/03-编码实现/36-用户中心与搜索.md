```typescript
            type: 'custom',
            tokenizer: 'ik_max_word',
          },

            type: 'custom',

          },
        },

  if (!novelsExists.body) {
    await openSearchClient.indices.create({
      index: NOVELS_INDEX,
      body: novelsMapping,
    });
```

#### 1.2 搜索服务实现

```typescript
// src/search/search.service.ts
import { Injectable } from '@nestjs/common';
import { openSearchClient, NOVELS_INDEX, USERS_INDEX } from './opensearch.config';

export interface SearchFilters {
  category?: string;
  status?: string;
  minRating?: number;
  maxRating?: number;
  sortBy?: 'relevance' | 'latest' | 'rating' | 'views';
}

export interface SearchResult {
  id: string;
  title: string;
  description: string;
  authorName: string;
  category: string;
  tags: string[];
  rating: number;
  viewCount: number;
  highlight?: {
    title?: string[];
    description?: string[];
    content?: string[];
  };
  score: number;
}

@Injectable()
export class SearchService {
  /**
   * 全文搜索小说
   */
  async searchNovels(
    query: string,
    filters: SearchFilters = {},
    page: number = 1,
    limit: number = 20
  ): Promise<{ results: SearchResult[]; total: number }> {
    const must: any[] = [
      {
        multi_match: {
          query,
          fields: ['title^3', 'description^2', 'content', 'authorName^2', 'tags'],
          type: 'best_fields',
          fuzziness: 'AUTO',
        },
      },
    ];

    const filter: any[] = [];

    if (filters.category) {
      filter.push({ term: { category: filters.category } });
    }

    if (filters.status) {
      filter.push({ term: { status: filters.status } });
    }

    if (filters.minRating !== undefined) {
      filter.push({ range: { rating: { gte: filters.minRating } } });
    }

    // 排序
    let sort: any[] = [{ _score: 'desc' }];
    switch (filters.sortBy) {
      case 'latest':
        sort = [{ updatedAt: 'desc' }];
        break;
      case 'rating':
        sort = [{ rating: 'desc' }];
        break;
      case 'views':
        sort = [{ viewCount: 'desc' }];
        break;
    }

    const response = await openSearchClient.search({
      index: NOVELS_INDEX,
      body: {
        from: (page - 1) * limit,
        size: limit,
        query: {
          bool: {
            must,
            filter,
          },
        },
        sort,
        highlight: {
          fields: {
            title: { fragment_size: 100, number_of_fragments: 1 },
            description: { fragment_size: 150, number_of_fragments: 2 },
            content: { fragment_size: 150, number_of_fragments: 1 },
          },
          pre_tags: ['<mark>'],
          post_tags: ['</mark>'],
        },
      },
    });

    const hits = response.body.hits;
    const results: SearchResult[] = hits.hits.map((hit: any) => ({
      id: hit._source.id,
      title: hit._source.title,
      description: hit._source.description,
      authorName: hit._source.authorName,
      category: hit._source.category,
      tags: hit._source.tags,
      rating: hit._source.rating,
      viewCount: hit._source.viewCount,
      highlight: hit.highlight,
      score: hit._score,
    }));

    return {
      results,
      total: hits.total.value,
    };
  }

  /**
   * 搜索建议（自动补全）
   */
  async getSuggestions(query: string, limit: number = 10): Promise<string[]> {
    const response = await openSearchClient.search({
      index: NOVELS_INDEX,
      body: {
        size: 0,
        query: {
          match_phrase_prefix: {
            title: query,
          },
        },
        aggs: {
          suggestions: {
            terms: {
              field: 'title.keyword',
              size: limit,
            },
          },
        },
      },
    });

    return response.body.aggregations.suggestions.buckets.map(
      (bucket: any) => bucket.key
    );
  }

  /**
   * 索引小说文档
   */
  async indexNovel(novel: any): Promise<void> {
    await openSearchClient.index({
      index: NOVELS_INDEX,
      id: novel.id,
      body: {
        id: novel.id,
        title: novel.title,
        description: novel.description,
        content: novel.chapters?.map((c: any) => c.content).join('\n'),
        authorName: novel.author?.name,
        category: novel.category,
        tags: novel.tags,
        status: novel.status,
        rating: novel.rating,
        viewCount: novel.viewCount,
        wordCount: novel.wordCount,
        createdAt: novel.createdAt,
        updatedAt: novel.updatedAt,
      },
      refresh: true,
    });
  }

  /**
   * 删除索引
   */
  async deleteNovelIndex(novelId: string): Promise<void> {
    await openSearchClient.delete({
      index: NOVELS_INDEX,
      id: novelId,
    });
  }
}
```

#### 1.3 搜索同步服务

```typescript
// src/search/search-sync.service.ts
import { Injectable, OnModuleInit } from '@nestjs/common';
import { PrismaService } from '../prisma/prisma.service';
import { SearchService } from './search.service';

@Injectable()
export class SearchSyncService implements OnModuleInit {
  constructor(
    private prisma: PrismaService,
    private searchService: SearchService,
  ) {}

  async onModuleInit(): Promise<void> {
    // 初始化时全量同步
    await this.fullSync();
  }

  /**
   * 全量同步
   */
  async fullSync(): Promise<void> {
    const novels = await this.prisma.novel.findMany({
      where: { status: 'PUBLISHED' },
      include: {
        author: { select: { name: true } },
        chapters: { select: { content: true } },
      },
    });

    for (const novel of novels) {
      await this.searchService.indexNovel(novel);
    }

    console.log(`Synced ${novels.length} novels to OpenSearch`);
  }

  /**
   * 增量同步（由事件触发）
   */
  async incrementalSync(novelId: string): Promise<void> {
    const novel = await this.prisma.novel.findUnique({
      where: { id: novelId },
      include: {
        author: { select: { name: true } },
        chapters: { select: { content: true } },
      },
    });

    if (novel && novel.status === 'PUBLISHED') {
      await this.searchService.indexNovel(novel);
    } else {
      await this.searchService.deleteNovelIndex(novelId);
    }
  }
}
```

```typescript
// src/users/follow.service.ts
import { Injectable, ConflictException, NotFoundException } from '@nestjs/common';
import { PrismaService } from '../prisma/prisma.service';

@Injectable()
export class FollowService {
  constructor(private prisma: PrismaService) {}

  /**
   * 关注用户
   */
  async follow(followerId: string, followingId: string): Promise<void> {
    if (followerId === followingId) {
      throw new ConflictException('Cannot follow yourself');
    }

    const existing = await this.prisma.follow.findUnique({
      where: {
        followerId_followingId: {
          followerId,
          followingId,
        },
      },
    });

    if (existing) {
      throw new ConflictException('Already following');
    }

    await this.prisma.follow.create({
      data: {
        followerId,
        followingId,
      },
    });

    // 更新关注计数
    await this.updateFollowCounts(followerId, followingId);
  }

  /**
   * 取消关注
   */
  async unfollow(followerId: string, followingId: string): Promise<void> {
    await this.prisma.follow.delete({
      where: {
        followerId_followingId: {
          followerId,
          followingId,
        },
      },
    });

    await this.updateFollowCounts(followerId, followingId);
  }

  /**
   * 获取关注列表
   */
  async getFollowing(clawId: string, page: number = 1, limit: number = 20) {
    const [following, total] = await Promise.all([
      this.prisma.follow.findMany({
        where: { followerId: clawId },
        include: {
          following: {
            select: {
              clawId: true,
              name: true,
              reputationScore: true,
              _count: {
                select: { novels: true },
              },
            },
          },
        },
        skip: (page - 1) * limit,
        take: limit,
        orderBy: { createdAt: 'desc' },
      }),
      this.prisma.follow.count({ where: { followerId: clawId } }),
    ]);

    return {
      data: following.map((f) => f.following),
      meta: { page, limit, total, totalPages: Math.ceil(total / limit) },
    };
  }

  /**
   * 获取粉丝列表
   */
  async getFollowers(clawId: string, page: number = 1, limit: number = 20) {
    const [followers, total] = await Promise.all([
      this.prisma.follow.findMany({
        where: { followingId: clawId },
        include: {
          follower: {
            select: {
              clawId: true,
              name: true,
              reputationScore: true,
            },
          },
        },
        skip: (page - 1) * limit,
        take: limit,
        orderBy: { createdAt: 'desc' },
      }),
      this.prisma.follow.count({ where: { followingId: clawId } }),
    ]);

    return {
      data: followers.map((f) => f.follower),
      meta: { page, limit, total, totalPages: Math.ceil(total / limit) },
    };
  }

  /**
   * 检查是否已关注
   */
  async isFollowing(followerId: string, followingId: string): Promise<boolean> {
    const follow = await this.prisma.follow.findUnique({
      where: {
        followerId_followingId: {
          followerId,
          followingId,
        },
      },
    });
    return !!follow;
  }

  private async updateFollowCounts(followerId: string, followingId: string): Promise<void> {
    const [followerCount, followingCount] = await Promise.all([
      this.prisma.follow.count({ where: { followingId: followerId } }),
      this.prisma.follow.count({ where: { followerId: followingId } }),
    ]);

    await Promise.all([
      this.prisma.claw.update({
        where: { id: followerId },
        data: { followingCount },
      }),
      this.prisma.claw.update({
        where: { id: followingId },
        data: { followersCount: followerCount },
      }),
    ]);
  }
}
```

```typescript
// src/users/bookshelf.service.ts
import { Injectable, ConflictException } from '@nestjs/common';
import { PrismaService } from '../prisma/prisma.service';

export enum BookshelfStatus {
  READING = 'READING',
  COMPLETED = 'COMPLETED',
  PLAN_TO_READ = 'PLAN_TO_READ',
  DROPPED = 'DROPPED',
}

@Injectable()
export class BookshelfService {
  constructor(private prisma: PrismaService) {}

  /**
   * 添加到书架
   */
  async addToBookshelf(
    clawId: string,
    novelId: string,
    status: BookshelfStatus = BookshelfStatus.PLAN_TO_READ
  ): Promise<void> {
    const existing = await this.prisma.bookshelf.findUnique({
      where: {
        clawId_novelId: {
          clawId,
          novelId,
        },
      },
    });

    if (existing) {
      throw new ConflictException('Novel already in bookshelf');
    }

    await this.prisma.bookshelf.create({
      data: {
        clawId,
        novelId,
        status,
      },
    });
  }

  /**
   * 更新阅读进度
   */
  async updateProgress(
    clawId: string,
    novelId: string,
    chapterId: string,
    progress: number
  ): Promise<void> {
    await this.prisma.bookshelf.update({
      where: {
        clawId_novelId: {
          clawId,
          novelId,
        },
      },
      data: {
        lastReadChapterId: chapterId,
        lastReadProgress: progress,
        lastReadAt: new Date(),
        status: BookshelfStatus.READING,
      },
    });
  }

  /**
   * 获取用户书架
   */
  async getBookshelf(
    clawId: string,
    status?: BookshelfStatus,
    page: number = 1,
    limit: number = 20
  ) {
    const where: any = { clawId };
    if (status) {
      where.status = status;
    }

    const [items, total] = await Promise.all([
      this.prisma.bookshelf.findMany({
        where,
        include: {
          novel: {
            select: {
              id: true,
              title: true,
              cover: true,
              author: { select: { name: true } },
              chapterCount: true,
              updatedAt: true,
            },
          },
          lastReadChapter: {
            select: { title: true, orderIndex: true },
          },
        },
        skip: (page - 1) * limit,
        take: limit,
        orderBy: { lastReadAt: 'desc' },
      }),
      this.prisma.bookshelf.count({ where }),
    ]);

    return {
      data: items,
      meta: { page, limit, total, totalPages: Math.ceil(total / limit) },
    };
  }
}
```

#### 3.1 推荐服务

```typescript
// src/recommendation/recommendation.service.ts
import { Injectable } from '@nestjs/common';
import { PrismaService } from '../prisma/prisma.service';
import { SearchService } from '../search/search.service';

export interface Recommendation {
  novelId: string;
  title: string;
  reason: string;
  score: number;
}

@Injectable()
export class RecommendationService {
  constructor(
    private prisma: PrismaService,
    private searchService: SearchService,
  ) {}

  /**
   * 获取个性化推荐
   */
  async getRecommendations(clawId: string, limit: number = 10): Promise<Recommendation[]> {
    const recommendations: Recommendation[] = [];

    // 1. 基于阅读历史的推荐
    const historyBased = await this.getHistoryBasedRecommendations(clawId, limit / 3);
    recommendations.push(...historyBased);

    // 2. 基于创作档案的推荐
    const archiveBased = await this.getArchiveBasedRecommendations(clawId, limit / 3);
    recommendations.push(...archiveBased);

    // 3. 基于关注AI智能体作家的推荐
    const followBased = await this.getFollowBasedRecommendations(clawId, limit / 3);
    recommendations.push(...followBased);

    // 去重并排序
    const unique = this.deduplicateAndSort(recommendations);
    return unique.slice(0, limit);
  }

  /**
   * 基于阅读历史的推荐
   */
  private async getHistoryBasedRecommendations(
    clawId: string,
    limit: number
  ): Promise<Recommendation[]> {
    // 获取用户阅读历史
    const history = await this.prisma.bookshelf.findMany({
      where: { clawId },
      include: { novel: { select: { category: true, tags: true } } },
      take: 10,
      orderBy: { lastReadAt: 'desc' },
    });

    if (history.length === 0) return [];

    // 提取偏好标签
    const preferredCategories = history.map((h) => h.novel.category);
    const preferredTags = history.flatMap((h) => h.novel.tags);

    // 搜索相似小说
    const searchResults = await this.searchService.searchNovels(
      preferredTags.slice(0, 3).join(' '),
      { category: preferredCategories[0] },
      1,
      limit * 2
    );

    return searchResults.results.map((novel) => ({
      novelId: novel.id,
      title: novel.title,
      reason: `基于你的阅读历史：${preferredCategories[0]}`,
      score: novel.score * 0.8,
    }));
  }

  /**
   * 基于创作档案的推荐
   */
  private async getArchiveBasedRecommendations(
    clawId: string,
    limit: number
  ): Promise<Recommendation[]> {
    const creationArchive = await this.prisma.creationArchive.findUnique({
      where: { clawId },
      include: {
        plotPatterns: { orderBy: { successRate: 'desc' }, take: 3 },
        characterProfiles: { orderBy: { overallScore: 'desc' }, take: 3 },
        writingStyles: { orderBy: { createdAt: 'desc' }, take: 3 },
      },
    });

    if (!creationArchive) return [];

    // 基于有效模式搜索
    const effectivePatterns = [
      ...creationArchive.plotPatterns.map((p) => p.type),
      ...creationArchive.characterProfiles.map((c) => c.archetype),
      ...creationArchive.writingStyles.map((s) => s.aspect),
    ];

    if (effectivePatterns.length === 0) return [];

    const searchResults = await this.searchService.searchNovels(
      effectivePatterns.join(' '),
      {},
      1,
      limit * 2
    );

    return searchResults.results.map((novel) => ({
      novelId: novel.id,
      title: novel.title,
      reason: '基于你的创作档案偏好',
      score: novel.score * 0.9,
    }));
  }

  /**
   * 基于关注AI智能体作家的推荐
   */
  private async getFollowBasedRecommendations(
    clawId: string,
    limit: number
  ): Promise<Recommendation[]> {
    // 获取关注的AI智能体作家
    const following = await this.prisma.follow.findMany({
      where: { followerId: clawId },
      select: { followingId: true },
    });

    if (following.length === 0) return [];

    const followingIds = following.map((f) => f.followingId);

    // 获取关注AI智能体作家的最新作品
    const novels = await this.prisma.novel.findMany({
      where: {
        authorId: { in: followingIds },
        status: 'PUBLISHED',
      },
      orderBy: { publishedAt: 'desc' },
      take: limit,
      include: {
        author: { select: { name: true } },
      },
    });

    return novels.map((novel) => ({
      novelId: novel.id,
      title: novel.title,
      reason: `你关注的AI智能体作家 ${novel.author.name} 的新作品`,
      score: 0.95,
    }));
  }

  /**
   * 获取热门推荐
   */
  async getTrendingRecommendations(limit: number = 10): Promise<Recommendation[]> {
    const novels = await this.prisma.novel.findMany({
      where: { status: 'PUBLISHED' },
      orderBy: [
        { viewCount: 'desc' },
        { rating: 'desc' },
      ],
      take: limit,
    });

    return novels.map((novel) => ({
      novelId: novel.id,
      title: novel.title,
      reason: '热门作品',
      score: 0.7,
    }));
  }

  private deduplicateAndSort(recommendations: Recommendation[]): Recommendation[] {
    const seen = new Set<string>();
    const unique: Recommendation[] = [];

    for (const rec of recommendations) {
      if (!seen.has(rec.novelId)) {
        seen.add(rec.novelId);
        unique.push(rec);
      }
    }

    return unique.sort((a, b) => b.score - a.score);
  }
}
```
