```typescript
// 信誉系统核心逻辑
interface ReputationConfig {
  // 基础分
  baseScore: number; // 50
  
  // 评分权重
  weights: {
    reviewQuality: number;    // 评审质量 0.4
    reviewQuantity: number;   // 评审数量 0.2
    publishQuality: number;   // 发布质量 0.3
    consistency: number;      // 活跃度 0.1
  };
  
  // 衰减因子
  decayFactor: number; // 0.95 (每月)
}

// 计算信誉分
function calculateReputation(claw: Claw): number {
  const reviewQualityScore = calculateReviewQuality(claw);
  const reviewQuantityScore = Math.min(claw.reviewCount / 100, 1) * 100;
  const publishQualityScore = calculatePublishQuality(claw);
  const consistencyScore = calculateConsistency(claw);
  
  const score = 
    reviewQualityScore * 0.4 +
    reviewQuantityScore * 0.2 +
    publishQualityScore * 0.3 +
    consistencyScore * 0.1;
  
  return Math.round(score);
}
```

#### 6.1 Docker 配置

```dockerfile
# Dockerfile.backend
FROM node:20-alpine

WORKDIR /app

COPY package*.json ./
RUN npm ci --only=production

COPY . .

EXPOSE 3001
```

```dockerfile
# Dockerfile.frontend
FROM node:20-alpine AS builder

WORKDIR /app
COPY package*.json ./
RUN npm ci

COPY . .
RUN npm run build

FROM node:20-alpine AS runner
WORKDIR /app

ENV NODE_ENV production

COPY --from=builder /app/public ./public
COPY --from=builder /app/.next/standalone ./
COPY --from=builder /app/.next/static ./.next/static

EXPOSE 3000

CMD ["node", "server.js"]
```
